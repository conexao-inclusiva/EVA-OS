"""Classe base para todos os agentes da EVA."""

from typing import Dict, Any, Optional, List
from abc import ABC, abstractmethod
from datetime import datetime

from src.kernel.events import Event, EventBus


class BaseAgent(ABC):
    """
    Classe base para todos os agentes.
    
    Todo agente deve herdar desta classe e implementar:
    - initialize(): Inicialização
    - start(): Início das operações
    - process(): Processamento de tarefas
    - shutdown(): Desligamento
    - health_check(): Verificação de saúde
    """
    
    # Atributos de plugin (devem ser sobrescritos)
    __plugin_name__ = "base_agent"
    __plugin_version__ = "1.0.0"
    __plugin_description__ = "Agente base da EVA"
    __plugin_author__ = "EVA OS Team"
    __plugin_tags__ = ["agent", "base"]
    __plugin_capabilities__ = ["base"]
    __plugin_requires__ = ["event_bus", "logger"]
    
    def __init__(self, event_bus: EventBus, logger, config, name: str = None):
        self.event_bus = event_bus
        self.logger = logger
        self.config = config
        
        # Nome do agente
        self.name = name or self.__class__.__name__.lower().replace("agent", "")
        
        # Estado
        self._running = False
        self._initialized = False
        self._task_count = 0
        self._error_count = 0
        self._last_task: Optional[Dict[str, Any]] = None
        self._started_at: Optional[datetime] = None
        
        # Capacidades
        self.capabilities = self._get_capabilities()
    
    @abstractmethod
    async def initialize(self) -> None:
        """Inicializa o agente."""
        self._initialized = True
        self.logger.info(f"Agente {self.name} inicializado")
    
    @abstractmethod
    async def start(self) -> None:
        """Inicia o agente."""
        self._running = True
        self._started_at = datetime.now()
        self.logger.info(f"Agente {self.name} iniciado")
    
    @abstractmethod
    async def process(self, task: Dict[str, Any]) -> Dict[str, Any]:
        """
        Processa uma tarefa.
        
        Args:
            task: Dicionário com a tarefa a ser processada
            
        Returns:
            Dict com o resultado do processamento
        """
        pass
    
    @abstractmethod
    async def shutdown(self) -> None:
        """Desliga o agente."""
        self._running = False
        self.logger.info(f"Agente {self.name} desligado")
    
    @abstractmethod
    async def health_check(self) -> Dict[str, Any]:
        """Verifica a saúde do agente."""
        return {
            "name": self.name,
            "running": self._running,
            "initialized": self._initialized,
            "task_count": self._task_count,
            "error_count": self._error_count,
            "uptime": (datetime.now() - self._started_at).total_seconds() if self._started_at else 0
        }
    
    def _get_capabilities(self) -> List[str]:
        """Obtém as capacidades do agente."""
        return getattr(self, "__plugin_capabilities__", ["general"])
    
    def can_handle(self, intent: str) -> bool:
        """Verifica se o agente pode lidar com uma intenção."""
        # Por padrão, verifica se a intenção está nas capacidades
        return intent in self.capabilities or "general" in self.capabilities
    
    async def handle_event(self, event: Event) -> None:
        """Manipula um evento recebido."""
        if not self._running:
            self.logger.warning(f"Agente {self.name} recebeu evento mas está desligado")
            return
        
        try:
            # Extrai tarefa do evento
            task = event.payload.get("task", {})
            if not task:
                task = event.payload
            
            # Processa a tarefa
            result = await self.process(task)
            
            # Registra
            self._task_count += 1
            self._last_task = {
                "task": task,
                "result": result,
                "timestamp": datetime.now().isoformat()
            }
            
            # Publica resultado
            await self.event_bus.publish_async(
                "agent.task_completed",
                {
                    "agent": self.name,
                    "result": result,
                    "original_event": event.id if hasattr(event, 'id') else None,
                    "timestamp": datetime.now().isoformat()
                },
                source=self.name
            )
            
        except Exception as e:
            self._error_count += 1
            self.logger.error(f"Erro no agente {self.name}: {e}")
            
            # Publica erro
            await self.event_bus.publish_async(
                "agent.task_failed",
                {
                    "agent": self.name,
                    "error": str(e),
                    "task": event.payload,
                    "timestamp": datetime.now().isoformat()
                },
                source=self.name
            )
    
    def get_stats(self) -> Dict[str, Any]:
        """Obtém estatísticas do agente."""
        return {
            "name": self.name,
            "running": self._running,
            "initialized": self._initialized,
            "task_count": self._task_count,
            "error_count": self._error_count,
            "uptime": (datetime.now() - self._started_at).total_seconds() if self._started_at else 0,
            "capabilities": self.capabilities,
            "last_task": self._last_task
        }

from typing import Dict, Any, Optional
import subprocess
import platform
import psutil

from src.agents.base_agent import BaseAgent
from src.kernel.events import EventBus


class InformationAgent(BaseAgent):
    """
    Agente de Informação.
    
    Responsável por:
    - Informações do sistema
    - Status de serviços
    - Informações de rede
    - Dados de desempenho
    """
    
    __plugin_name__ = "information_agent"
    __plugin_version__ = "1.0.0"
    __plugin_description__ = "Agente especializado em informações do sistema"
    __plugin_author__ = "EVA OS Team"
    __plugin_tags__ = ["agent", "information", "system"]
    __plugin_capabilities__ = ["information", "system_info", "status"]
    __plugin_requires__ = ["event_bus", "logger", "config"]
    
    def __init__(self, event_bus: EventBus, logger, config):
        super().__init__(event_bus, logger, config, name="information")
    
    async def process(self, task: Dict[str, Any]) -> Dict[str, Any]:
        """
        Processa uma tarefa de informação.
        """
        task_type = task.get("type", "system")
        
        if task_type == "system":
            return await self._handle_system_info(task)
        elif task_type == "network":
            return await self._handle_network_info(task)
        elif task_type == "resources":
            return await self._handle_resources(task)
        elif task_type == "services":
            return await self._handle_services(task)
        else:
            return {"error": f"Tipo de tarefa desconhecido: {task_type}"}
    
    async def _handle_system_info(self, task: Dict[str, Any]) -> Dict[str, Any]:
        """Manipula informações do sistema."""
        return {
            "type": "system",
            "platform": platform.platform(),
            "python_version": platform.python_version(),
            "hostname": platform.node(),
            "processor": platform.processor(),
            "timestamp": __import__('datetime').datetime.now().isoformat()
        }
    
    async def _handle_network_info(self, task: Dict[str, Any]) -> Dict[str, Any]:
        """Manipula informações de rede."""
        # TODO: Implementar informações de rede completas
        return {
            "type": "network",
            "hostname": platform.node(),
            "timestamp": __import__('datetime').datetime.now().isoformat()
        }
    
    async def _handle_resources(self, task: Dict[str, Any]) -> Dict[str, Any]:
        """Manipula informações de recursos."""
        cpu = psutil.cpu_percent(interval=0.1)
        memory = psutil.virtual_memory()
        disk = psutil.disk_usage('/')
        
        return {
            "type": "resources",
            "cpu": cpu,
            "memory": {
                "percent": memory.percent,
                "used_gb": memory.used / (1024**3),
                "total_gb": memory.total / (1024**3)
            },
            "disk": {
                "percent": disk.percent,
                "used_gb": disk.used / (1024**3),
                "total_gb": disk.total / (1024**3)
            },
            "timestamp": __import__('datetime').datetime.now().isoformat()
        }
    
    async def _handle_services(self, task: Dict[str, Any]) -> Dict[str, Any]:
        """Manipula informações de serviços."""
        # TODO: Implementar informações de serviços
        return {
            "type": "services",
            "timestamp": __import__('datetime').datetime.now().isoformat()
        }
    
    async def shutdown(self) -> None:
        """Desliga o agente."""
        await super().shutdown()
    
    async def health_check(self) -> Dict[str, Any]:
        """Verifica saúde do agente."""
        return await super().health_check()
    async def initialize(self) -> None:
        if self._initialized:
            return
        await super().initialize()

    async def start(self) -> None:
        if self._running:
            return
        if not self._initialized:
            raise RuntimeError("Agente nao inicializado")
        await super().start()

    async def _on_user_message(self, event: Event) -> None:
        data = getattr(event, "data", None)
        if not isinstance(data, dict):
            self.logger.warning("[BRAIN] Evento USER_MESSAGE sem dados válidos")
            return
        text = data.get("text", "")
        user_id = str(data.get("user") or "unknown").strip() or "unknown"
        # Identificacao explicita do interlocutor.
        # Nao altera nem migra memorias antigas.
        import re as _re
        import unicodedata as _ud

        def _normalizar_identidade(valor):
            valor = _ud.normalize("NFKD", valor.casefold())
            return "".join(c for c in valor if not _ud.combining(c))

        identidade = _normalizar_identidade(text.strip())
        troca = _re.fullmatch(
            r"(?:eva[,! ]*)?(?:sou eu|aqui e|quem fala e)"
            r"[, :]+(?:o |a )?(alberto|juliana|nicolas|nick|liris)[.! ]*",
            identidade,
        )

        if troca:
            nomes = {
                "alberto": "alberto",
                "juliana": "juliana",
                "nicolas": "nicolas",
                "nick": "nicolas",
                "liris": "liris",
            }
            user_id = nomes[troca.group(1)]
            self.logger.info("[BRAIN] Interlocutor declarado: %s", user_id)

        # Mantem o interlocutor por sessao, sem compartilhar identidade
        # automaticamente entre conexoes diferentes.
        if not hasattr(self, "_session_speakers"):
            self._session_speakers = {}

        source = str(data.get("source") or getattr(event, "source", "unknown"))
        session = data.get("session_id")

        if isinstance(session, (str, int)) and str(session).strip():
            session_key = (source, str(session))
        else:
            session_key = None

        if troca:
            if session_key is not None:
                self._session_speakers[session_key] = user_id
        elif session_key is not None:
            user_id = self._session_speakers.get(session_key, user_id)

        user_manager = self._modules.get("user_manager")
        if user_manager is not None:
            try:
                profile = await user_manager.process({
                    "user_id": user_id,
                    "message": text,
                })
                user_id = profile.get("user_id", user_id)
            except Exception:
                self.logger.exception("[BRAIN] Falha ao atualizar perfil do usuario")

        if not isinstance(text, str) or not text.strip():
            return
        text = text.strip()
        self.logger.info(f"[BRAIN] Mensagem de {user_id}: {text[:50]!r}")
        if text.casefold() in ("stop", "eva stop"):
            voice = self._modules.get("voice_manager")
            if voice and hasattr(voice, "stop_listening"):
                try:
                    result = voice.stop_listening()
                    if inspect.isawaitable(result):
                        await result
                    await self._say("Escuta desativada. Diga EVA para ativar novamente.")
                except Exception:
                    self.logger.exception("[BRAIN] Falha ao interromper escuta")
            return
        coordinator = self.conversation_coordinator
        turn_id = None
        if coordinator is not None:
            try:
                turn_id = resolve_turn_id(event)
                received = coordinator.receive_turn(
                    turn_id=turn_id,
                    user_id=user_id,
                    content=text,
                    session_id=str(session) if session is not None else None,
                    source=source,
                    timestamp=getattr(event, 'timestamp', datetime.now()).isoformat(),
                )
                if not received:
                    self.logger.info('[BRAIN] Turno repetido: %s', turn_id)
                    return
                coordinator.start_turn(turn_id)
            except Exception:
                self.logger.exception('[BRAIN] Falha ao registrar turno')
                return
        started = False
        try:
            await self.event_bus.publish_async(
                EventType.PROCESSING_START,
                {"message": text, "user": user_id, "timestamp": datetime.now().isoformat()},
                source="brain",
            )
            started = True
            response = await self._handle_pending_family_correction(text, user_id)
            if response is None:
                response = await self._handle_pending_relational_correction(text, user_id)
            if response is None:
                response = await self._handle_pending_general_correction(text, user_id)
            if response is None:
                response = await self._handle_pending_name_correction(text, user_id)
            if response is None:
                response = await self._try_family_removal(text, user_id)
            if response is None:
                response = await self._try_family_learning(text, user_id)
            if response is None:
                response = await self._try_relational_learning(text, user_id)
            if response is None:
                response = await self._try_general_learning(text, user_id)
            if response is None:
                response = await self._try_local_name_learning(text, user_id)
            if response is None:
                response = await self._generate_response(text, user_id)
            if not isinstance(response, str):
                response = str(response) if response is not None else "Não consegui gerar uma resposta."
            await self._persist_turn(user_id, text, response)
            if coordinator is not None:
                coordinator.complete_turn(
                    turn_id,
                    response,
                    timestamp=datetime.now().isoformat(),
                )
            await self._persist_facts(user_id, text, response)
            await self._say(response)
            await self.event_bus.publish_async(
                EventType.ASSISTANT_RESPONSE,
                {"text": response, "user": user_id, "timestamp": datetime.now().isoformat()},
                source="brain",

            self.logger.exception("[BRAIN] Falha ao processar mensagem")
        finally:
            if started:
                try:

    # ===== MEMÓRIA =====
    
    async def _retrieve_memories(self, text: str, user_id: str, context_data: Dict) -> List[Dict]:
        memories = []
        try:
            memory_manager = self._modules.get("memory_manager")
            if memory_manager:
                if hasattr(memory_manager, "_search_relevant_memories"):
                    memories = memory_manager._search_relevant_memories(user_id, text) or []

    async def _handle_pending_family_correction(self, text: str, user_id: str) -> Optional[str]:

                       family_relations.norm(fact['object'])]

                return 'Essa relação já está registrada na minha memória.'
            if result == 'saved':
                self.logger.info('[BRAIN] Relacao familiar armazenada: %s -> %s -> %s',
                    fact['subject'], fact['relation'], fact['object'])
                return 'Registrei a relação na memória familiar: ' + fact['object'] + '.'

    def _resolve_family_fact(self, text: str, user_id: str) -> Optional[str]:
        lookup = family_relations.question(text)
        if lookup is None or self.family_relations is None:
            return None
        subject, relation, template = lookup
        relations = ('filho', 'filha') if relation == 'filhos' else (relation,)
        rows = []
        for rel in relations:
            rows.extend(self.family_relations.active(user_id, subject, rel))
        if not rows:
            return (
                'Ainda nao tenho essa relacao familiar confirmada '
                'na minha memoria. Voce pode me informar quem e?'
            )
        values = []
        seen = set()
        for row in rows:
            value = row['object']
            key = family_relations.norm(value)
            if key not in seen:
                seen.add(key)
                values.append(value)
        if relation in family_relations.SINGULAR and len(values) != 1:
            return None
        if relation in {'filho', 'filha'} and len(values) > 1:
            return ('Seus filhos registrados são ' if relation == 'filho' else
                    'Suas filhas registradas são ') + ', '.join(values) + '.'
        return template.format(', '.join(values))

    async def _resolve_unified_vehicles(self, text: str, user_id: str) -> Optional[str]:
        """Une veículos explicitamente registrados nos dois bancos, sem inferências.

        Não migra dados, não grava fatos e não modifica o carro atual.
        """
        normalized_question = fact_learning.normalize(
            fact_learning.strip_eva(text).strip(' .?!')
        )
        if normalized_question not in {
            'quais carros eu tenho', 'quais veiculos eu tenho',
            'que carros eu tenho', 'que veiculos eu tenho',
        }:
            return None

        vehicles = []
        seen = set()
        current = None

        long_term = self._modules.get('long_term_memory')
        if long_term is not None and hasattr(long_term, 'get_all_facts'):
            facts = await long_term.get_all_facts(user_id)
            candidates = [f for f in facts
                          if f.get('source') == 'ensinado_geral_explicitamente'
                          and fact_learning.normalize(f.get('subject', '')) == 'veiculo'
                          and f.get('relation') == 'carro_atual']
            if not candidates:
                candidates = [f for f in facts
                              if f.get('source') == 'ensinado_explicitamente'
                              and fact_learning.normalize(f.get('subject', '')) == 'carro'
                              and f.get('relation') == 'nome']
            distinct = {fact_learning.normalize(f.get('object', '')) for f in candidates
                        if str(f.get('object', '')).strip()}
            # Não escolher arbitrariamente um carro atual em caso de conflito.
            if len(distinct) == 1:
                current = str(candidates[0]['object']).strip()
                seen.add(fact_learning.normalize(current))
                vehicles.append(current)

        if self.relational_memory is not None:
            rows = self.relational_memory.active(user_id, 'usuario', 'possui')
            for row in rows:
                value = str(row.get('object', '')).strip()
                key = fact_learning.normalize(value)
                if value and key not in seen:
                    seen.add(key)
                    vehicles.append(value)

        if not vehicles:
            return None
        if current:
            others = [v for v in vehicles if fact_learning.normalize(v) !=
                      fact_learning.normalize(current)]
            if others:
                return (f'Seu carro atual registrado é {current}. Você também '
                        f'registrou: {", ".join(others)}.')
            return f'Seu carro atual registrado é {current}.'
        return 'Você registrou estes veículos: ' + ', '.join(vehicles) + '.'

                   and fact_learning.normalize(f.get('subject', '')) == subject
                   and f.get('relation') == relation]
        values = {fact_learning.normalize(f['object']) for f in matches}

    async def _try_local_name_learning(self, text: str, user_id: str) -> Optional[str]:
        """Aprende nomes confirmados sem LLM; nao altera fatos divergentes."""
        # Fatos de veiculos, projetos e preferencias pertencem ao extrator geral.
        # Nao os reinterpretar como "nome" caso o armazenamento geral falhe.
        if fact_learning.extract(text) is not None:
            return None
        fact = self._extract_explicit_fact(text)
        if not fact:
            return None
        long_term = self._modules.get("long_term_memory")
        if not long_term or not all(hasattr(long_term, method) for method in ("get_all_facts", "store_fact")):
            return None
        try:
            existing = await long_term.get_all_facts(user_id)
            same_key = [item for item in existing if
                        self._normalize_fact_text(item.get("subject", "")) == fact["subject"] and
                        self._normalize_fact_text(item.get("relation", "")) == "nome" and
                        item.get("source") == "ensinado_explicitamente"]
            different = [item for item in same_key if
                         self._normalize_fact_text(item.get("object", "")) !=
                         self._normalize_fact_text(fact["object"])]
            if different:
                self._pending_name_corrections[user_id] = {
                    "subject": fact["subject"],
                    "object": fact["object"],
                    "old_values": sorted({item["object"] for item in same_key}),
                }
                self.logger.warning("[BRAIN] Nome divergente; aguardando confirmacao: %s", fact["subject"])
                return (f"Eu tinha registrado outro nome para {fact['subject']}. "
                        f"Você quer corrigir para {fact['object']}? Ainda não alterei minha memória.")
            if not same_key:
                fact_id = await long_term.store_fact(
                    user_id=user_id, subject=fact["subject"], relation="nome",
                    obj=fact["object"], confidence=0.95, source="ensinado_explicitamente",
                )
                if fact_id == -1:
                    return None
            self.logger.info("[BRAIN] Nome aprendido localmente sem LLM: %s", fact["subject"])
            return f"Entendi. O nome do seu {fact['subject']} é {fact['object']}. Vou lembrar disso."
        except Exception:
            self.logger.exception("[BRAIN] Falha no aprendizado local de nome")
            return None

    def _extract_explicit_fact(self, text: str) -> Optional[Dict[str, str]]:
        """Extrai somente atribuicoes explicitas de nome, sem inferir fatos."""
        cleaned = re.sub(r"^\s*(?:eva|ei eva|oi eva)\s*[,!:]?\s*", "", text, flags=re.I).strip()
        # Evita interpretar perguntas e negacoes como informacoes confirmadas.
        if "?" in cleaned or re.search(r"\b(?:nao e|não é|talvez|acho que)\b", cleaned, re.I):
            return None
        explicit = re.match(
            r"^(?:quero que voce guarde esta informacao|quero que você guarde esta informação|"
            r"guarde esta informacao|guarde esta informação|lembre-se de que|"
            r"lembre que|anote que|guarde que)\s*[:,-]?\s*", cleaned, re.I,
        )
        # Aceita declaracoes naturais estritas alem de comandos explicitos.
        statement = cleaned[explicit.end():].strip() if explicit else cleaned
        if not explicit and re.search(
            r"\b(?:penso|pensando|talvez|possivelmente|pretendo|planejo|seria|fosse|" 
            r"queria|gostaria|acho|nao|não)\b", statement, re.I,
        ):
            return None
        match = re.fullmatch(
            r"(?:o nome do meu|o nome da minha|meu|minha)\s+"
            r"(?P<subject>[\wÀ-ÿ][\wÀ-ÿ -]{1,65}?)\s+"
            r"(?:é|e|se chama)\s+(?P<value>[^.!?\n]{2,100})[.!]?",
            statement, re.I,
        )
        if not match:
            return None
        subject = self._normalize_fact_text(match.group("subject"))
        value = match.group("value").strip().strip('"“”\' ').rstrip('.')
        if not subject or not value or self._normalize_fact_text(value) in {"nao sei", "desconhecido"}:
            return None
        return {"subject": subject, "relation": "nome", "object": value}

    async def _resolve_persistent_fact(self, text: str, user_id: str) -> Optional[str]:
        """Responde perguntas estritas sobre nomes gravados pelo usuario."""
        long_term = self._modules.get("long_term_memory")
        if not long_term or not hasattr(long_term, "get_all_facts"):
            return None
        question = re.sub(r"^\s*(?:eva|ei eva|oi eva)\s*[,!:]?\s*", "", text, flags=re.I).strip()
        match = re.fullmatch(
            r"(?:qual (?:é|e) o nome d[oa] (?:meu|minha)|"
            r"como se chama (?:o meu|a minha))\s+"
            r"(?P<subject>[\wÀ-ÿ][\wÀ-ÿ -]{1,65}?)\s*[?.!]*",
            question, re.I,
        )
        if not match:
            return None
        subject = self._normalize_fact_text(match.group("subject"))
        facts = await long_term.get_all_facts(user_id)
        matches = [fact for fact in facts if
                   self._normalize_fact_text(fact.get("subject", "")) == subject and
                   self._normalize_fact_text(fact.get("relation", "")) == "nome" and
                   fact.get("source") == "ensinado_explicitamente"]
        # Nao adivinhar quando houver valores conflitantes.
        unique = {self._normalize_fact_text(fact.get("object", "")) for fact in matches}
        if len(unique) != 1 or not matches:
            return None
        answer = str(matches[0]["object"]).strip()
        if not answer:
            return None
        self.logger.info("[BRAIN] Fato explicito recuperado da LongTermMemory sem LLM")
        return f"O nome do seu {match.group('subject').strip()} é {answer}."

    def _resolve_clock_request(self, text: str) -> Optional[str]:
        """Responde apenas perguntas inequivocas sobre o momento atual.

        Nao interpreta horarios historicos, de outras cidades ou compromissos.
        """
        time_manager = self._modules.get("time_manager")
        if time_manager is None:
            return None

        normalized = text.strip().casefold()
        normalized = re.sub(r"[?.!]+$", "", normalized).strip()
        normalized = re.sub(r"^(?:eva|ei eva|oi eva)[,!: ]+", "", normalized).strip()
        normalized = re.sub(r"\s+", " ", normalized)

        if normalized in {
            "que horas são", "que horas sao", "que horas são agora",
            "que horas sao agora", "qual é a hora", "qual e a hora",
            "qual é a hora agora", "qual e a hora agora",
            "me diga as horas", "me diz as horas", "pode me dizer as horas",
        }:
            return f"Agora são {time_manager.get_time()}."

        if normalized in {
            "qual é a data de hoje", "qual e a data de hoje",
            "que dia é hoje", "que dia e hoje", "qual a data de hoje",
            "qual é a data atual", "qual e a data atual",
        }:
            return f"Hoje é {time_manager.get_date()}."

        if normalized in {
            "qual é o dia da semana", "qual e o dia da semana",
            "que dia da semana é hoje", "que dia da semana e hoje",
            "qual é o dia da semana hoje", "qual e o dia da semana hoje",
        }:
            return f"Hoje é {time_manager.get_weekday().lower()}."

        return None

    async def _generate_local_response(self, text: str, user_id: str, context: Dict) -> Optional[str]:


        # Consulta contextual: pronome resolvido + fato explicito.
        import re as _ctx_re
        import unicodedata as _ctx_ud

        def _ctx_normalize(value):
            value = _ctx_ud.normalize("NFKD", str(value or ""))
            value = "".join(
                c for c in value if not _ctx_ud.combining(c)
            )
            return _ctx_re.sub(r"\s+", " ", value.casefold()).strip()

        _ctx_question = _ctx_normalize(text).rstrip("?.! ")
        _ctx_match = _ctx_re.fullmatch(
            r"(?:qual (?:e|era) o nome(?: de)?|como (?:se chama|se chamava)) "
            r"(dele|dela)",
            _ctx_question,
        )

        if _ctx_match and isinstance(context, dict):
            _ctx_payload = context.get("references", {})
            if (
                isinstance(_ctx_payload, dict)
                and isinstance(_ctx_payload.get("references"), dict)
            ):
                _ctx_refs = _ctx_payload["references"]
                _ctx_unresolved = _ctx_payload.get("unresolved", [])
            else:
                _ctx_refs = _ctx_payload
                _ctx_unresolved = context.get("unresolved_references", [])

            if (
                isinstance(_ctx_refs, dict)
                and isinstance(_ctx_unresolved, list)
                and not _ctx_unresolved
            ):
                _ctx_pronoun = _ctx_match.group(1)
                _ctx_entities = [
                    entity
                    for phrase, entity in _ctx_refs.items()
                    if _ctx_normalize(phrase) == _ctx_pronoun
                    and isinstance(entity, dict)
                    and entity.get("name")
                ]

                if len(_ctx_entities) == 1:
                    _ctx_subject = _ctx_normalize(
                        _ctx_entities[0]["name"]
                    )
                    _ctx_memory = self._modules.get("long_term_memory")

                    if _ctx_memory and hasattr(
                        _ctx_memory, "get_all_facts"
                    ):
                        try:
                            _ctx_facts = _ctx_memory.get_all_facts(user_id)
                            if inspect.isawaitable(_ctx_facts):
                                _ctx_facts = await _ctx_facts

                            _ctx_values = set()

                            for _ctx_fact in _ctx_facts or []:
                                if not isinstance(_ctx_fact, dict):
                                    continue

                                _ctx_relation = _ctx_normalize(
                                    _ctx_fact.get("relation")
                                )
                                _ctx_confidence = _ctx_fact.get(
                                    "confidence", 0
                                )
                                _ctx_value = str(
                                    _ctx_fact.get("object") or ""
                                ).strip()

                                if (
                                    _ctx_normalize(
                                        _ctx_fact.get("subject")
                                    ) == _ctx_subject
                                    and _ctx_relation in (
                                        "nome", "name", "nome_completo"
                                    )
                                    and isinstance(
                                        _ctx_confidence, (int, float)
                                    )
                                    and _ctx_confidence >= 0.9
                                    and _ctx_value
                                ):
                                    _ctx_values.add(_ctx_value)

                            if len(_ctx_values) == 1:
                                return next(iter(_ctx_values))

                        except Exception:
                            self.logger.exception(
                                "[BRAIN] Falha na consulta contextual"
                            )

        # Ponte contextual familiar: exige referencia unica e registro confirmado.
        if _ctx_match and isinstance(context, dict):
            _family_refs = context.get("references", {})
            _family_unresolved = context.get("unresolved_references", [])

            if (
                isinstance(_family_refs, dict)
                and isinstance(_family_unresolved, list)
                and not _family_unresolved
                and self.family_relations is not None
            ):
                _pronoun = _ctx_match.group(1)
                _entity = _family_refs.get(_pronoun)

                if isinstance(_entity, dict) and _entity.get("name"):
                    _person = _ctx_normalize(_entity["name"])
                    _family_candidates = set()

                    for _relation in ("pai", "mae", "filho", "filha", "irmao", "irma"):
                        try:
                            _rows = self.family_relations.active(
                                user_id, "usuario", _relation
                            )
                        except Exception:
                            self.logger.exception(
                                "[BRAIN] Falha ao consultar parentesco contextual"
                            )
                            _rows = []

                        for _row in _rows or []:
                            if not isinstance(_row, dict):
                                continue
                            _value = str(_row.get("object") or "").strip()
                            if _value and _ctx_normalize(_value) == _person:
                                _family_candidates.add(_value)

                    if len(_family_candidates) == 1:
                        return next(iter(_family_candidates))

        # Consulta factual conservadora, sem LLM.
        # Apenas perguntas explicitas sobre nome, com correspondencia exata.
        import re
        import unicodedata

        def normalize_name(value):
            value = unicodedata.normalize("NFKD", str(value or ""))
            value = "".join(c for c in value if not unicodedata.combining(c))
            return re.sub(r"\s+", " ", value.casefold()).strip()

        question = normalize_name(text).rstrip("?.! ")
        match = re.fullmatch(
            r"(?:qual (?:e|era) o nome de|como (?:se chama|se chamava)) "
            r"(?P<person>.+)",
            question,
        )

        if match:
            long_term = self._modules.get("long_term_memory")
            if long_term and hasattr(long_term, "get_all_facts"):
                try:
                    facts = long_term.get_all_facts(user_id)
                    if inspect.isawaitable(facts):
                        facts = await facts

                    person = match.group("person")
                    candidates = []

                    for fact in facts or []:
                        if not isinstance(fact, dict):
                            continue

                        subject = normalize_name(fact.get("subject"))
                        relation = normalize_name(fact.get("relation"))
                        value = str(fact.get("object") or "").strip()
                        confidence = fact.get("confidence", 0)

                        if (
                            subject == person
                            and relation in ("nome", "name", "nome_completo")
                            and value
                            and isinstance(confidence, (int, float))
                            and confidence >= 0.9
                        ):
                            candidates.append(value)

                    unique = set(candidates)
                    if len(unique) == 1:
                        return next(iter(unique))

                except Exception:
                    self.logger.exception(
                        "[BRAIN] Falha na consulta factual local"
                    )

        knowledge_manager = self._modules.get("knowledge_manager")
        if knowledge_manager:
            try:
                if hasattr(knowledge_manager, "query"):
                    result = knowledge_manager.query(text)
                    if inspect.isawaitable(result):
                        result = await result
                    if (isinstance(result, dict) and isinstance(result.get("answer"), str)
                            and isinstance(result.get("confidence"), (int, float))
                            and 0.8 <= result["confidence"] <= 1.0):
                        return result["answer"]
            except Exception:
                self.logger.exception("[BRAIN] Erro ao consultar conhecimento local")
        reasoning_manager = self._modules.get("reasoning_manager")
        if reasoning_manager:
            try:
                if hasattr(reasoning_manager, "reason"):
                    result = reasoning_manager.reason(message=text, context=context)
                    if inspect.isawaitable(result):
                        result = await result
                    if (isinstance(result, dict) and isinstance(result.get("answer"), str)
                            and isinstance(result.get("confidence"), (int, float))
                            and 0.8 <= result["confidence"] <= 1.0):
                        return result["answer"]
            except Exception:
                self.logger.exception("[BRAIN] Erro no raciocínio local")
        return None
    
    # ===== SAUDAÇÃO =====
    
    def _detect_explicit_greeting(self, text: str) -> bool:
        """Verifica se a mensagem contém uma saudação explícita."""
        greetings = [
            r'\bolá\b', r'\boi\b', r'\be aí\b', r'\bfala\b', r'\bopa\b',
            r'\bbom dia\b', r'\bboa tarde\b', r'\bboa noite\b',
            r'\beai\b', r'\biae\b', r'\bsalve\b'
        ]
        pattern = '|'.join(greetings)
        return bool(re.search(pattern, text, re.IGNORECASE))
    
    async def _should_greet(self, user_id: str, explicit_greeting: bool) -> bool:
        """Decide se deve cumprimentar o usuário."""
        # Se já cumprimentou na sessão, não repete (a menos que seja saudação explícita)
        if self._greeting_done.get(user_id, False):
            self.logger.debug(f"🧠 [BRAIN] Já cumprimentou {user_id} na sessão")
            return False  # Não repete
        
        # Se é saudação explícita, sempre cumprimenta
        if explicit_greeting:
            self.logger.info(f"🧠 [BRAIN] Saudação explícita para {user_id}")
            self._greeting_done[user_id] = True
            return True
        
        # Verifica se é a primeira interação do dia
        long_term = self._modules.get("long_term_memory")
        if long_term:
            try:
                last_date_str = await long_term.get_last_greeting(user_id)
                if last_date_str:
                    last_date = datetime.fromisoformat(last_date_str).date()
                    today = datetime.now().date()
                    if last_date < today:
                        self.logger.info(f"🧠 [BRAIN] Primeira interação do dia para {user_id}")
                        self._greeting_done[user_id] = True
                        return True
                    else:
                        self.logger.debug(f"🧠 [BRAIN] Já saudou hoje para {user_id}")
                        # Mesmo que já tenha saudado hoje, se for primeira vez na sessão, pode saudar?
                        # Vamos deixar como: se já saudou hoje e não é explícito, não repete.
                        return False
                else:
                    # Nunca saudou antes
                    self.logger.info(f"🧠 [BRAIN] Primeira saudação para {user_id}")
                    self._greeting_done[user_id] = True
                    return True
            except Exception as e:
                self.logger.error(f"🧠 [BRAIN] Erro ao verificar última saudação: {e}")
                return False
        return False

                context = context_raw
                context.update(context_data)
            else:
                context = context_data
            
            # ============================================================
            # 🔥 CORREÇÃO: GARANTIR QUE context['state']['resolved_references'] SEJA dict
            # ============================================================
            if isinstance(context, dict):
                state = context.get("state", {})
                if isinstance(state, dict):
                    refs = state.get("resolved_references", {})
                    if not isinstance(refs, dict):
                        self.logger.warning(f"🧠 [BRAIN] 'resolved_references' é {type(refs)}. Convertendo para dict.")
                        if isinstance(refs, list):
                            try:
                                refs = dict(refs)
                            except:
                                refs = {}
                        else:
                            refs = {}
                        state["resolved_references"] = refs
                        context["state"] = state
            
            # Preserva entidades no formato original (lista ou dict).
            # O KnowledgeManager revisado aceita ambos os formatos.

            # ============================================================
            # MEMÓRIA
            # ============================================================
            if memory_manager and hasattr(memory_manager, "process"):
                memory = await memory_manager.process(context)
                if not isinstance(memory, dict):
                    self.logger.warning(f"⚠️ memory_manager retornou {type(memory)}. Convertendo para dict.")
                    memory = {"history": memory if isinstance(memory, list) else []}
            else:
                memory = {"history": []}
            
            # ============================================================
            # CONHECIMENTO (COM VALIDAÇÃO E LOGS)
            # ============================================================
            if knowledge_manager and hasattr(knowledge_manager, "process"):
                knowledge = await knowledge_manager.process(context, memory)
                self.logger.debug(f"📊 [BRAIN] knowledge type: {type(knowledge)}")
                if not isinstance(knowledge, dict):
                    self.logger.warning(f"⚠️ knowledge_manager retornou {type(knowledge)}. Convertendo para dict.")
                    if isinstance(knowledge, list):
                        knowledge = {"facts": knowledge, "keywords": []}
                    else:
                        knowledge = {"facts": []}
                else:
                    self.logger.debug(f"📊 [BRAIN] knowledge keys: {list(knowledge.keys())}")
            else:
                knowledge = {"facts": []}
            
            # Resposta local: somente usa uma resposta explícita do módulo.
            # Resultados de busca, isoladamente, não são uma resposta pronta.
            local_response = await self._generate_local_response(text, user_id, context)
            decision_gate = self.llm_decision_gate or self._modules.get("llm_decision_gate")
            use_local = False
            if isinstance(local_response, str) and local_response.strip():
                if decision_gate:
                    try:
                        use_local = not await decision_gate.should_call_llm(
                            context, {"response": local_response, "confidence": 0.8, "resolved": True}
                        )
                    except TypeError:
                        # Compatibilidade com versões antigas do DecisionGate:
                        # não libera resposta local sem validação explícita.
                        self.logger.warning("[BRAIN] DecisionGate antigo; usando LLM")
                    except Exception:
                        self.logger.exception("[BRAIN] Falha ao validar resposta local")
                else:
                    self.logger.debug("[BRAIN] Sem DecisionGate; usando LLM")
            if use_local:
                self.logger.info("[BRAIN] Resposta local validada; sem chamada ao LLM")
                return local_response.strip()

            # ============================================================
            # PLANEJAMENTO (COM VALIDAÇÃO)
            # ============================================================
            if planner_manager and hasattr(planner_manager, "process"):
                plan = await planner_manager.process(context, memory, knowledge)
                if not isinstance(plan, dict):
                    self.logger.warning(f"⚠️ planner_manager retornou {type(plan)}. Convertendo para dict.")
                    plan = {"plan": {"name": "conversation", "steps": ["responder"]}, "strategy": "default"}
            else:
                plan = {"plan": {"name": "conversation", "steps": ["responder"]}, "strategy": "default"}
            
            # ============================================================
            # RACIOCÍNIO (COM VALIDAÇÃO)
            # ============================================================
            if reasoning_manager and hasattr(reasoning_manager, "process"):
                reasoning = await reasoning_manager.process(context, memory, knowledge, plan)
                if not isinstance(reasoning, dict):
                    self.logger.warning(f"⚠️ reasoning_manager retornou {type(reasoning)}. Convertendo para dict.")
                    reasoning = {"inferences": {}, "conclusions": [], "recommendations": []}
            else:
                reasoning = {"conclusions": []}
            
            # ============================================================
            # EMOÇÃO (COM VALIDAÇÃO)
            # ============================================================
            if emotion_engine and hasattr(emotion_engine, "process"):
                emotion = await emotion_engine.process(context, memory, knowledge, plan, reasoning)
                if not isinstance(emotion, dict):
                    self.logger.warning(f"⚠️ emotion_engine retornou {type(emotion)}. Convertendo para dict.")
                    emotion = {"tone": "neutro"}
            else:
                emotion = {"tone": "neutro"}
            
            # ============================================================
            # PROMPT
            # ============================================================
            if prompt_builder and hasattr(prompt_builder, "process"):
                prompt_data = await prompt_builder.process(context, memory, knowledge, plan, reasoning, emotion)
                prompt = prompt_data.get("prompt", text) if isinstance(prompt_data, dict) else text
            else:
                prompt = text
            
            # ============================================================
            # LLM
            # ============================================================
            if llm:
                import time
                inicio_llm = time.perf_counter()
                self.logger.info(f"[BRAIN] Iniciando LLM | Prompt: {len(prompt)} caracteres")
                try:
                    response = await llm.generate(prompt)
                finally:
                    tempo_llm = time.perf_counter() - inicio_llm
                    self.logger.info(f"[BRAIN] Tempo LLM: {tempo_llm:.2f} segundos")
                
                # 🔥 ATUALIZA ESTADO DO CONTEXT ENGINE (SEGURAMENTE)
                if context_engine and hasattr(context_engine, "state") and hasattr(context_engine.state, "update"):
                    try:
                        if isinstance(response, str):
                            context_engine.state.update({"last_response": response})
                    except Exception as e:
                        self.logger.warning(f"🧠 [BRAIN] Erro ao atualizar estado do ContextEngine: {e}")
                
                return response
            
            return f"Recebi: '{text[:50]}...' (LLM não disponível)"
            
        except Exception as e:
            self.logger.error(f"🧠 [BRAIN] Erro no pipeline: {e}")
            import traceback
            self.logger.error(f"🧠 [BRAIN] Traceback: {traceback.format_exc()}")
            if self._llm_manager:
                try:
                    return await self._llm_manager.generate(text)
                except Exception:
                    self.logger.exception("[BRAIN] Fallback LLM falhou")
                    return "Desculpe, tive um problema ao processar sua mensagem."
            return f"Recebi sua mensagem: '{text}'"

class PluginManager:
    """
    Gerenciador de Plugins da EVA.
    
    Características:
    - Descoberta automática de módulos
    - Resolução de dependências
    - Carregamento dinâmico
    - Hot reload
    - Sandbox (futuro)
    - Verificação de assinatura (futuro)
    - Compatibilidade entre versões
    """
    
    def __init__(self, logger, event_bus, config, service_registry):
        self.logger = logger
        self.event_bus = event_bus
        self.config = config
        self.registry = service_registry
        
        # Plugins
        self._delegated_plugins = {}
        self._plugin_load_locks = {}
        self._plugins: Dict[str, Plugin] = {}
        self._plugins_by_name: Dict[str, str] = {}  # nome -> id
        
        # Caminhos de scan
        self._scan_paths: List[Path] = []
        
        # Padrões de exclusão
        self._exclude_patterns = {
            "__pycache__",
            "tests",
            "test",
            "venv",
            ".venv",
            "env",
            ".git",
            ".idea",
            ".vscode",
            "node_modules",
            "dist",
            "build"
        }
        
        # Plugins ignorados
        self._ignored_plugins: Set[str] = set()
        
        # Plugins carregados na ordem
        self._load_order: List[str] = []
        
        # Configuração
        self._auto_load = config.get("features.auto_load_plugins", True)
        self._enable_hot_reload = config.get("features.hot_reload", False)
        
        # Estado
        self._is_scanning = False
        self._is_loading = False
        
        self.logger.info("PluginManager inicializado")
    
    def add_scan_path(self, path: Path) -> None:
        """Adiciona um caminho para scanear."""
        if path.exists() and path.is_dir():
            self._scan_paths.append(path.resolve())
            self.logger.debug(f"Caminho adicionado: {path}")
        else:
            self.logger.warning(f"Caminho inválido: {path}")
    
    def add_exclude_pattern(self, pattern: str) -> None:
        """Adiciona um padrão de exclusão."""
        self._exclude_patterns.add(pattern)
    
    def ignore_plugin(self, name: str) -> None:
        """Ignora um plugin específico."""
        self._ignored_plugins.add(name)
        self.logger.debug(f"Plugin ignorado: {name}")
    
    async def scan(self) -> Dict[str, Plugin]:
        """
        Escaneia todos os caminhos em busca de plugins.
        Retorna dicionário de plugins encontrados.
        """
        if self._is_scanning:
            self.logger.warning("Scan já em andamento")
            return self._plugins
        
        self._is_scanning = True
        self.logger.info("Iniciando scan de plugins...")
        
        found = {}
        
        for path in self._scan_paths:
            plugins = await self._scan_directory(path)
            found.update(plugins)
        
        self._is_scanning = False
        self.logger.info(f"Scan concluído: {len(found)} plugins encontrados")
        
        # Atualiza plugins
        for plugin_id, plugin in found.items():
            if plugin_id not in self._plugins:
                self._plugins[plugin_id] = plugin
                self._plugins_by_name[plugin.manifest.name] = plugin_id
            else:
                # Atualiza existente
                existing = self._plugins[plugin_id]
                existing.status = plugin.status
                existing.path = plugin.path
        
        return self._plugins
    
    async def _scan_directory(self, path: Path, parent: Optional[Path] = None) -> Dict[str, Plugin]:
        """Escaneia um diretório recursivamente."""
        plugins = {}
        
        try:
            for item in path.iterdir():
                # Verifica exclusões
                if item.name in self._exclude_patterns:
                    continue
                
                if item.name.startswith("."):
                    continue
                
                if item.is_dir():
                    # Verifica se é um pacote Python (tem __init__.py)
                    init_file = item / "__init__.py"
                    if init_file.exists():
                        # É um pacote - procura por plugins dentro
                        sub_plugins = await self._scan_directory(item, path)
                        plugins.update(sub_plugins)
                    else:
                        # Diretório normal - pode ter arquivos .py
                        for py_file in item.glob("*.py"):
                            if py_file.name.startswith("__"):
                                continue
                            plugin = await self._discover_plugin(py_file, item)
                            if plugin:
                                plugins[plugin.manifest.id] = plugin
                
                elif item.is_file() and item.suffix == ".py":
                    if item.name.startswith("__"):
                        continue
                    plugin = await self._discover_plugin(item, parent or path)
                    if plugin:
                        plugins[plugin.manifest.id] = plugin
        
        except Exception as e:
            self.logger.error(f"Erro ao scanear {path}: {e}")
        
        return plugins
    
    async def _discover_plugin(self, file_path: Path, base_path: Path) -> Optional[Plugin]:
        """Descobre metadados sem instanciar o plugin."""
        try:
            file_path = file_path.resolve()
            module_name = self._get_module_name_from_path(file_path)
            if (module_name in self._ignored_plugins
                    or file_path.stem in self._ignored_plugins):
                return None
            try:
                module = importlib.import_module(module_name)
            except Exception as exc:
                self.logger.debug(
                    "Falha ao importar " + module_name + ": " + type(exc).__name__
                )
                return None

            imported_path = getattr(module, "__file__", None)
            if imported_path is None or Path(imported_path).resolve() != file_path:
                self.logger.warning("Origem de modulo divergente: " + module_name)
                return None

            plugin_class = self._find_plugin_class(module)
            if plugin_class is None:
                return None
            manifest = self._create_manifest_from_class(
                plugin_class, module_name, file_path
            )
            if manifest is None or manifest.name in self._ignored_plugins:
                return None
            return Plugin(
                manifest=manifest,
                path=file_path,
                module=module,
                status=PluginStatus.DISCOVERED,
            )
        except Exception as exc:
            self.logger.debug(
                "Falha na descoberta de " + str(file_path) + ": " + type(exc).__name__
            )
            return None

    
    def _find_plugin_class(self, module: Any) -> Optional[Type]:
        """Seleciona uma classe concreta definida pelo proprio modulo."""
        candidates = []
        for name, obj in inspect.getmembers(module, inspect.isclass):
            if name.startswith("_"):
                continue
            if obj.__module__ != module.__name__ or inspect.isabstract(obj):
                continue
            if not (hasattr(obj, "initialize") or hasattr(obj, "start")):
                continue
            if (hasattr(obj, "__plugin_name__")
                    or hasattr(obj, "__plugin_version__")):
                candidates.append(obj)
                continue
            try:
                parameters = list(inspect.signature(obj.__init__).parameters.values())[1:]
                required = [
                    parameter for parameter in parameters
                    if parameter.default is inspect.Parameter.empty
                    and parameter.kind not in (
                        inspect.Parameter.VAR_POSITIONAL,
                        inspect.Parameter.VAR_KEYWORD,
                    )
                ]
                if len(required) <= 2:
                    candidates.append(obj)
            except (TypeError, ValueError):
                continue

        module_leaf = module.__name__.rsplit(".", 1)[-1]
        matching = [
            cls for cls in candidates
            if cls.__dict__.get("__plugin_name__") == module_leaf
        ]
        if len(matching) == 1:
            return matching[0]
        if len(matching) > 1:
            self.logger.warning("Classes de plugin ambiguas: " + module.__name__)
            return None

        declared = [
            cls for cls in candidates
            if "__plugin_name__" in cls.__dict__
            or "__plugin_version__" in cls.__dict__
        ]
        remaining = declared or candidates
        if len(remaining) == 1:
            return remaining[0]
        if len(remaining) > 1:
            self.logger.warning("Classes de plugin ambiguas: " + module.__name__)
        return None

    
    def _create_manifest_from_class(self, cls: Type, module_name: str, file_path: Path) -> Optional[PluginManifest]:
        """Cria um manifesto a partir de uma classe."""
        try:
            manifest = PluginManifest(
                name=getattr(cls, "__plugin_name__", module_name),
                version=getattr(cls, "__plugin_version__", "1.0.0"),
                description=getattr(cls, "__plugin_description__", ""),
                author=getattr(cls, "__plugin_author__", "EVA OS"),
                main_class=cls.__name__,
                plugin_type=self._detect_plugin_type(cls, module_name)
            )
            
            # Detecta dependências
            if hasattr(cls, "__plugin_requires__"):
                manifest.requires = cls.__plugin_requires__
            
            # Detecta tags
            if hasattr(cls, "__plugin_tags__"):
                manifest.tags = cls.__plugin_tags__
            
            # Detecta capabilities
            if hasattr(cls, "__plugin_capabilities__"):
                manifest.capabilities = cls.__plugin_capabilities__
            
            return manifest
            
        except Exception as e:
            self.logger.error(f"Erro ao criar manifesto para {cls.__name__}: {e}")
            return None
    
    def _detect_plugin_type(self, cls: Type, module_name: str) -> PluginType:
        """Detecta o tipo do plugin baseado no nome e classe."""
        type_map = {
            "agent": PluginType.AGENT,
            "tool": PluginType.TOOL,
            "interface": PluginType.INTERFACE,
            "core": PluginType.CORE,
            "memory": PluginType.MODULE,
            "knowledge": PluginType.MODULE,
            "planner": PluginType.MODULE,
            "consciousness": PluginType.MODULE
        }
        
        # Verifica por nome
        module_lower = module_name.lower()
        for key, plugin_type in type_map.items():
            if key in module_lower:
                return plugin_type
        
        # Verifica por métodos
        if hasattr(cls, "execute") or hasattr(cls, "run"):
            return PluginType.TOOL
        
        if hasattr(cls, "process") or hasattr(cls, "decide"):
            return PluginType.AGENT
        
        return PluginType.MODULE
    
    async def load_all(self) -> Dict[str, bool]:
        """
        Carrega todos os plugins descobertos.
        Resolve dependências e carrega na ordem correta.
        """
        if self._is_loading:
            self.logger.warning("Carregamento já em andamento")
            return {}
        
        self._is_loading = True
        self.logger.info("Carregando plugins...")
        
        results = {}
        
        # 1. Resolve dependências
        self._resolve_dependencies()
        
        # 2. Ordena topologicamente
        self._load_order = self._topological_sort()
        
        # 3. Carrega na ordem
        for plugin_id in self._load_order:
            plugin = self._plugins.get(plugin_id)
            if plugin:
                success = await self._load_plugin(plugin)
                results[plugin.manifest.name] = success
                
                # Publica evento
                await self.event_bus.publish_async(
                    "PLUGIN_LOADED" if success else "PLUGIN_LOAD_FAILED",
                    {
                        "plugin": plugin.manifest.name,
                        "version": plugin.manifest.version,
                        "success": success,
                        "error": plugin.error if plugin.error else None
                    },
                    source="plugin_manager"
                )
        
        self._is_loading = False
        self.logger.info(f"Carregamento concluído: {sum(results.values())}/{len(results)} plugins")
        
        return results
    
    async def _load_plugin_owned(self, plugin: Plugin) -> bool:
        """Carrega um plugin específico."""
        if plugin.status in [PluginStatus.LOADED, PluginStatus.INITIALIZED, PluginStatus.STARTED]:
            return True
        
        if plugin.manifest.name in self._ignored_plugins:
            plugin.status = PluginStatus.STOPPED
            plugin.error = "Plugin ignorado"
            return False
        
        try:
            plugin.status = PluginStatus.LOADING

            # Nao substitui servicos pertencentes a outro carregamento.
            if self.registry.exists(plugin.manifest.name):
                plugin.status = PluginStatus.ERROR
                plugin.error = "Servico ja registrado; propriedade nao transferida"
                self.logger.error("Conflito de servico: " + plugin.manifest.name)
                return False
            
            # Verifica dependências
            for dep_name in plugin.manifest.requires:
                if self._resolve_service(dep_name) is not None:
                    continue
                if dep_name not in self._plugins_by_name:
                    plugin.status = PluginStatus.ERROR
                    plugin.error = f"Dependência não encontrada: {dep_name}"
                    self.logger.error(f"Plugin {plugin.manifest.name}: {plugin.error}")
                    return False
                
                dep_id = self._plugins_by_name[dep_name]
                dep = self._plugins.get(dep_id)
                if dep and dep.status not in [PluginStatus.LOADED, PluginStatus.INITIALIZED, PluginStatus.STARTED]:
                    plugin.status = PluginStatus.ERROR
                    plugin.error = f"Dependência não carregada: {dep_name}"
                    self.logger.error(f"Plugin {plugin.manifest.name}: {plugin.error}")
                    return False
            
            # Importa o módulo se ainda não foi
            if not plugin.module:
                # Reconstrói nome do módulo
                module_name = self._get_module_name_from_path(plugin.path)
                plugin.module = importlib.import_module(module_name)
            
            # Encontra a classe principal
            if plugin.manifest.main_class and plugin.module:
                cls = getattr(plugin.module, plugin.manifest.main_class, None)
                if cls:
                    # Instancia a classe
                    instance = await self._instantiate_plugin(cls, plugin)
                    if instance:
                        plugin.instance = instance
                        plugin.status = PluginStatus.LOADED
                        plugin.loaded_at = datetime.now()
                        
                        self.logger.info(f"Plugin carregado: {plugin.manifest.name} v{plugin.manifest.version}")
                        
                        # Registra no ServiceRegistry
                        self.registry.register(
                            plugin.manifest.name,
                            instance,
                            plugin.manifest.requires
                        )
                        
                        # Inicializa
                        if hasattr(instance, "initialize"):
                            try:
                                if asyncio.iscoroutinefunction(instance.initialize):
                                    await instance.initialize()
                                else:
                                    instance.initialize()
                                plugin.status = PluginStatus.INITIALIZED
                                plugin.initialized_at = datetime.now()
                            except Exception as e:
                                self.logger.error(f"Erro ao inicializar {plugin.manifest.name}: {e}")
                                plugin.error = str(e)
                                plugin.status = PluginStatus.ERROR
                                return False
                        
                        # Inicia
                        if hasattr(instance, "start"):
                            try:
                                if asyncio.iscoroutinefunction(instance.start):
                                    await instance.start()
                                else:
                                    instance.start()
                                plugin.status = PluginStatus.STARTED
                                plugin.started_at = datetime.now()
                            except Exception as e:
                                self.logger.error(f"Erro ao iniciar {plugin.manifest.name}: {e}")
                                plugin.error = str(e)
                                plugin.status = PluginStatus.ERROR
                                return False
                        
                        return True
            
            plugin.status = PluginStatus.ERROR
            plugin.error = "Classe principal não encontrada"
            return False
            
        except Exception as e:
            plugin.status = PluginStatus.ERROR
            plugin.error = str(e)
            self.logger.error(f"Erro ao carregar plugin {plugin.manifest.name}: {e}")
            return False


    
    async def _instantiate_plugin(self, cls: Type, plugin: Plugin) -> Optional[Any]:
        """Injeta primeiro pelo nome do parametro, depois pelo tipo registrado."""
        try:
            signature = inspect.signature(cls.__init__)
            args = []
            kwargs = {}
            for name, parameter in list(signature.parameters.items())[1:]:
                if parameter.kind in (
                    inspect.Parameter.VAR_POSITIONAL,
                    inspect.Parameter.VAR_KEYWORD,
                ):
                    continue

                value = self._resolve_service(name)
                if value is None:
                    annotation = parameter.annotation
                    type_name = getattr(annotation, "__name__", None)
                    if type_name:
                        value = self._resolve_service(type_name.lower())

                if value is None:
                    if parameter.default is inspect.Parameter.empty:
                        raise TypeError("Dependencia obrigatoria nao resolvida: " + name)
                    if parameter.kind == inspect.Parameter.POSITIONAL_ONLY:
                        args.append(parameter.default)
                    continue

                if parameter.kind == inspect.Parameter.POSITIONAL_ONLY:
                    args.append(value)
                else:
                    kwargs[name] = value

            return cls(*args, **kwargs)
        except Exception as exc:
            self.logger.error(
                "Erro ao instanciar " + cls.__name__ + ": " + type(exc).__name__
            )
            return None

    
    def _get_module_name_from_path(self, path: Path) -> str:
        """Preserva o pacote raiz no nome de importacao."""
        path = path.resolve()
        for root in path.parents:
            if root.name in ("src", "eva_os"):
                relative = path.relative_to(root.parent).with_suffix("")
                parts = list(relative.parts)
                if parts[-1] == "__init__":
                    parts.pop()
                if not parts or not all(part.isidentifier() for part in parts):
                    raise ValueError("Caminho de modulo invalido")
                return ".".join(parts)

        # Compatibilidade com outros pacotes Python explicitamente escaneados.
        parts = [] if path.stem == "__init__" else [path.stem]
        parent = path.parent
        while (parent / "__init__.py").is_file():
            parts.insert(0, parent.name)
            parent = parent.parent
        if not parts or not all(part.isidentifier() for part in parts):
            raise ValueError("Caminho de modulo invalido")
        return ".".join(parts)

    
    def _resolve_dependencies(self) -> None:
        """Inclui a propriedade dos agentes na ordem de carregamento."""
        owner_id = self._plugins_by_name.get("agent_manager")
        for plugin in self._plugins.values():
            dependencies = []
            for name in plugin.manifest.requires:
                dependency_id = self._plugins_by_name.get(name)
                if dependency_id and dependency_id not in dependencies:
                    dependencies.append(dependency_id)
            if (owner_id and self._is_managed_agent_plugin(plugin)
                    and owner_id not in dependencies):
                dependencies.append(owner_id)
            plugin.dependencies = dependencies
            plugin.dependents = []

        for plugin in self._plugins.values():
            for dependency_id in plugin.dependencies:
                dependency = self._plugins.get(dependency_id)
                if dependency is not None:
                    dependency.dependents.append(plugin.manifest.id)

    
    def _topological_sort(self) -> List[str]:
        """
        Ordena plugins topologicamente por dependências.
        Plugins com menos dependências são carregados primeiro.
        """
        # Kahn's algorithm
        graph = {}
        in_degree = {}
        
        for plugin_id, plugin in self._plugins.items():
            graph[plugin_id] = plugin.dependencies.copy()
            in_degree[plugin_id] = len(plugin.dependencies)
        
        # Fila com plugins que não têm dependências
        queue = [pid for pid, deg in in_degree.items() if deg == 0]
        result = []
        
        while queue:
            # Ordena por tipo (Core primeiro)
            queue.sort(key=lambda pid: 0 if self._plugins[pid].is_core() else 1)
            plugin_id = queue.pop(0)
            result.append(plugin_id)
            
            # Remove dependências de outros
            for other_id, deps in graph.items():
                if plugin_id in deps:
                    deps.remove(plugin_id)
                    in_degree[other_id] -= 1
                    if in_degree[other_id] == 0:
                        queue.append(other_id)
        
        # Verifica ciclos
        if len(result) != len(self._plugins):
            self.logger.warning("Ciclo de dependência detectado!")
            # Tenta carregar mesmo assim (ordem atual)
            return list(self._plugins.keys())
        
        return result
    
    async def reload(self, plugin_name: str) -> bool:
        """Recarrega um plugin específico (hot reload)."""
        if not self._enable_hot_reload:
            self.logger.warning("Hot reload desabilitado")
            return False
        
        if plugin_name not in self._plugins_by_name:
            self.logger.warning(f"Plugin não encontrado: {plugin_name}")
            return False
        
        plugin_id = self._plugins_by_name[plugin_name]
        plugin = self._plugins[plugin_id]

        if (plugin_id in self._delegated_plugins
                or any(owner is plugin.instance
                       for owner in self._delegated_plugins.values())):
            self.logger.warning("Reload requer coordenacao com o proprietario")
            return False
        
        # Para o plugin
        if plugin.instance and hasattr(plugin.instance, "shutdown"):
            try:
                if asyncio.iscoroutinefunction(plugin.instance.shutdown):
                    await plugin.instance.shutdown()
                else:
                    plugin.instance.shutdown()
            except Exception as e:
                self.logger.error(f"Erro ao parar {plugin_name}: {e}")
        
        # Recarrega o módulo
        if plugin.module:
            try:
                importlib.reload(plugin.module)
            except Exception as e:
                self.logger.error(f"Erro ao recarregar módulo {plugin_name}: {e}")
                return False
        
        # Recarrega o plugin
        plugin.status = PluginStatus.DISCOVERED
        plugin.loaded_at = None
        plugin.initialized_at = None
        plugin.started_at = None
        plugin.instance = None
        
        return await self._load_plugin(plugin)

    
    def get_plugin(self, name: str) -> Optional[Plugin]:
        """Obtém um plugin pelo nome."""
        plugin_id = self._plugins_by_name.get(name)
        return self._plugins.get(plugin_id) if plugin_id else None
    
    def get_plugins(self, status: Optional[PluginStatus] = None) -> List[Plugin]:
        """Lista plugins, opcionalmente filtrando por status."""
        if status:
            return [p for p in self._plugins.values() if p.status == status]
        return list(self._plugins.values())
    
    def get_status_report(self) -> Dict[str, Any]:
        """Gera relatório de status dos plugins."""
        return {
            "total": len(self._plugins),
            "by_status": {
                status.name: len([p for p in self._plugins.values() if p.status == status])
                for status in PluginStatus
            },
            "plugins": {
                name: {
                    "version": p.manifest.version,
                    "status": p.status.name,
                    "type": p.manifest.plugin_type.name,
                    "error": p.error,
                    "dependencies": p.manifest.requires,
                    "load_time": p.load_time
                }
                for name, p in self._plugins.items()
            },
            "load_order": self._load_order,
            "scan_paths": [str(p) for p in self._scan_paths]
        }
    
    async def shutdown(self) -> None:
        """Desliga somente instancias cujo ciclo de vida pertence ao carregador."""
        self.logger.info("Desligando plugins...")
        for plugin_id in reversed(self._load_order):
            if plugin_id in self._delegated_plugins:
                continue
            plugin = self._plugins.get(plugin_id)
            if plugin is None or plugin.instance is None:
                continue
            if plugin.status == PluginStatus.STOPPED:
                continue
            if not hasattr(plugin.instance, "shutdown"):
                continue
            try:
                if asyncio.iscoroutinefunction(plugin.instance.shutdown):
                    await plugin.instance.shutdown()
                else:
                    plugin.instance.shutdown()
                plugin.status = PluginStatus.STOPPED
            except Exception as exc:
                plugin.error = type(exc).__name__
                plugin.status = PluginStatus.ERROR
                self.logger.error(
                    "Erro ao parar " + plugin.manifest.name + ": " + type(exc).__name__
                )

        # Reflete a parada efetuada pelo proprietario, sem repetir shutdown.
        for plugin_id in self._delegated_plugins:
            plugin = self._plugins.get(plugin_id)
            if (plugin is not None and plugin.instance is not None
                    and getattr(plugin.instance, "_running", None) is False):
                plugin.status = PluginStatus.STOPPED
        self.logger.info("Plugins desligados")


    def _resolve_service(self, name: str) -> Optional[Any]:
        """Preserva o registro existente e reconhece dependencias do Core."""
        if self.registry.exists(name):
            return self.registry.get(name)
        provided = {
            "event_bus": self.event_bus,
            "logger": self.logger,
            "config": self.config,
            "service_registry": self.registry,
        }
        return provided.get(name)

    def _is_managed_agent_plugin(self, plugin: Plugin) -> bool:
        return (
            plugin.manifest.plugin_type == PluginType.AGENT
            and plugin.path.parent.name == "agents"
            and plugin.manifest.name.endswith("_agent")
            and plugin.manifest.name != "agent_manager"
        )

    async def _load_plugin(self, plugin: Plugin) -> bool:
        """Serializa a carga e respeita o proprietario da instancia."""
        lock = self._plugin_load_locks.setdefault(
            plugin.manifest.id, asyncio.Lock()
        )
        async with lock:
            return await self._load_plugin_coordinated(plugin)

    async def _load_plugin_coordinated(self, plugin: Plugin) -> bool:
        name = plugin.manifest.name

        def reject(message):
            plugin.status = PluginStatus.ERROR
            plugin.error = message
            self.logger.error("Plugin " + name + ": " + message)
            return False

        if name in self._ignored_plugins:
            plugin.status = PluginStatus.STOPPED
            plugin.error = "Plugin ignorado"
            return False

        if self._is_managed_agent_plugin(plugin):
            owner = self.registry.get("agent_manager")
            owner_discovered = "agent_manager" in self._plugins_by_name
            if owner is None and owner_discovered:
                return reject("AgentManager descoberto mas ainda indisponivel")

            if owner is not None:
                getter = getattr(owner, "get_agent", None)
                if (not callable(getter)
                        or not getattr(owner, "_running", False)
                        or getattr(owner, "_shutting_down", False)):
                    return reject("Proprietario de agentes indisponivel")
                try:
                    module = plugin.module
                    if module is None:
                        module = importlib.import_module(
                            self._get_module_name_from_path(plugin.path)
                        )
                    expected_class = getattr(module, plugin.manifest.main_class, None)
                    agent = getter(name[:-len("_agent")])
                    if (not inspect.isclass(expected_class)
                            or agent is None
                            or not isinstance(agent, expected_class)
                            or not getattr(agent, "_initialized", False)
                            or not getattr(agent, "_running", False)):
                        return reject("Agente nao iniciado pelo AgentManager")

                    # Dependencias continuam obrigatorias ao reutilizar a instancia.
                    for dependency_name in plugin.manifest.requires:
                        if self._resolve_service(dependency_name) is not None:
                            continue
                        dependency_id = self._plugins_by_name.get(dependency_name)
                        dependency = self._plugins.get(dependency_id)
                        if dependency is None or dependency.status != PluginStatus.STARTED:
                            return reject("Dependencia indisponivel: " + dependency_name)

                    if self.registry.exists(name):
                        if self.registry.get(name) is not agent:
                            return reject("Servico existente pertence a outra instancia")
                    else:
                        self.registry.register(name, agent, plugin.manifest.requires)

                    plugin.module = module
                    plugin.instance = agent
                    plugin.status = PluginStatus.STARTED
                    plugin.error = None
                    if plugin.loaded_at is None:
                        plugin.loaded_at = datetime.now()
                    plugin.started_at = getattr(agent, "_started_at", None)
                    self._delegated_plugins[plugin.manifest.id] = owner
                    return True
                except Exception as exc:
                    return reject("Falha ao vincular agente: " + type(exc).__name__)

        # Repeticao de carga propria nao cria nem inicia outra instancia.
        if (plugin.status == PluginStatus.STARTED
                and plugin.instance is not None
                and self.registry.get(name) is plugin.instance):
            return True

        if name == "agent_manager":
            # Nao transfere agentes que ja foram iniciados individualmente.
            from src.agents.base_agent import BaseAgent
            for service in self.registry.list_services():
                if isinstance(service.instance, BaseAgent):
                    return reject(
                        "Agentes individuais ja registrados; transferencia exige coordenacao"
                    )

        if plugin.manifest.id in self._delegated_plugins:
            return reject("Propriedade anterior nao pode ser transferida automaticamente")
        return await self._load_plugin_owned(plugin)

"""Gerenciador de geração via Ollama para EVA-OS.

O histórico e o contexto são preparados pelo Brain/PromptBuilder. Este módulo
não guarda conversas entre chamadas para evitar duplicação e mistura de usuários.
"""
import asyncio
from typing import Any, Dict, List, Optional
import aiohttp
from src.kernel.events import EventBus


class LLMManager:
    def __init__(self, event_bus: EventBus, logger, config):
        self.event_bus = event_bus
        self.logger = logger
        self.config = config
        self._ollama_host = config.get("llm.ollama.host", "localhost")
        self._ollama_port = config.get("llm.ollama.port", 11434)
        self._model = config.get("llm.model", "qwen2.5:7b")
        self._running = False
        # Mantido por compatibilidade com consumidores que consultem o atributo.
        self._conversation_history: List[Dict[str, str]] = []
        self.logger.info(f"LLMManager configurado com modelo: {self._model}")

    def _base_url(self) -> str:
        host = str(self._ollama_host).rstrip("/")
        if host.startswith(("http://", "https://")):
            return host if host.rsplit(":", 1)[-1].isdigit() else f"{host}:{self._ollama_port}"
        return f"http://{host}:{self._ollama_port}"

    async def initialize(self) -> None:
        self.logger.info("LLMManager inicializando...")
        self._running = True

        for tentativa in range(1, 4):
            if await self._check_ollama():
                break

            if tentativa < 3:
                self.logger.warning(
                    f"Ollama indispon?vel em {self._base_url()} "
                    f"(tentativa {tentativa}/3). Nova tentativa em 1 segundo."
                )
                await asyncio.sleep(1)
        else:
            self._running = False
            raise RuntimeError(
                f"Ollama n?o respondeu em {self._base_url()} ap?s 3 tentativas."
            )

        models = await self._get_available_models()
        if models and self._model not in models:
            self.logger.warning(
                f"Modelo configurado '{self._model}' n?o encontrado. "
                f"Dispon?veis: {', '.join(models)}. N?o trocando automaticamente."
            )

        self.logger.info(f"LLMManager inicializado com modelo: {self._model}")

    async def generate(self, prompt: str, system_prompt: Optional[str] = None) -> str:
        if not isinstance(prompt, str) or not prompt.strip():
            return "Não recebi uma mensagem válida para processar."
        url = f"{self._base_url()}/api/chat"
        data = {
            "model": self._model,
            "messages": self._build_messages(prompt, system_prompt),
            "stream": False,
            "options": {"temperature": 0.8, "top_p": 0.9, "num_predict": 1024},
        }
        try:
            async with aiohttp.ClientSession() as session:
                async with session.post(url, json=data, timeout=aiohttp.ClientTimeout(total=120)) as resp:
                    if resp.status != 200:
                        details = (await resp.text())[:300]
                        self.logger.error(f"Ollama HTTP {resp.status}: {details}")
                        return self._fallback_response(prompt)
                    result = await resp.json()

                    # Metricas de desempenho retornadas pelo Ollama
                    total = result.get("total_duration", 0) / 1e9
                    load = result.get("load_duration", 0) / 1e9
                    prompt_time = result.get("prompt_eval_duration", 0) / 1e9
                    eval_time = result.get("eval_duration", 0) / 1e9
                    prompt_tokens = result.get("prompt_eval_count", 0)
                    generated_tokens = result.get("eval_count", 0)
                    speed = generated_tokens / eval_time if eval_time > 0 else 0

                    self.logger.info(
                        f"[LLM METRICAS] Total: {total:.2f}s | "
                        f"Carregamento: {load:.2f}s | "
                        f"Prompt: {prompt_time:.2f}s ({prompt_tokens} tokens) | "
                        f"Geracao: {eval_time:.2f}s ({generated_tokens} tokens) | "
                        f"Velocidade: {speed:.2f} tokens/s"
                    )

                    answer = result.get("message", {}).get("content", "")
                    if isinstance(answer, str) and answer.strip():
                        return answer.strip()
                    self.logger.warning("Ollama retornou resposta vazia")
                    return self._fallback_response(prompt)
        except (asyncio.TimeoutError, aiohttp.ServerTimeoutError):
            self.logger.error("Timeout ao aguardar resposta do Ollama")
            return "Desculpe, o modelo demorou demais para responder. Pode tentar novamente?"
        except (aiohttp.ClientError, OSError, ValueError) as exc:
            self.logger.error(f"Falha na comunicação com Ollama: {exc}")
            return self._fallback_response(prompt)

    def _build_messages(self, prompt: str, system_prompt: Optional[str] = None) -> list:
        system = system_prompt if system_prompt is not None else (
            "Você é EVA, uma assistente virtual familiar. Responda em português "
            "brasileiro com clareza, naturalidade e respeito. Não invente fatos "
            "sobre pessoas ou sobre a família; quando não souber, diga isso."
        )
        return [
            {"role": "system", "content": system},
            {"role": "user", "content": prompt},
        ]

    def _fallback_response(self, prompt: str) -> str:
        return "Desculpe, não consegui processar sua mensagem agora. Pode repetir, por favor?"

    async def _check_ollama(self) -> bool:
        try:
            async with aiohttp.ClientSession() as session:
                async with session.get(
                    f"{self._base_url()}/api/tags", timeout=aiohttp.ClientTimeout(total=3)
                ) as resp:
                    return resp.status == 200
        except (aiohttp.ClientError, asyncio.TimeoutError, OSError):
            return False

    async def _get_available_models(self) -> List[str]:
        try:
            async with aiohttp.ClientSession() as session:
                async with session.get(
                    f"{self._base_url()}/api/tags", timeout=aiohttp.ClientTimeout(total=3)
                ) as resp:
                    if resp.status == 200:
                        data = await resp.json()
                        return [m["name"] for m in data.get("models", [])
                                if isinstance(m, dict) and isinstance(m.get("name"), str)]
        except (aiohttp.ClientError, asyncio.TimeoutError, OSError, ValueError) as exc:
            self.logger.warning(f"Não foi possível listar modelos: {exc}")
        return []

    def clear_history(self) -> None:
        self._conversation_history.clear()
        self.logger.info("Histórico interno do LLM limpo (contexto gerenciado pelo Brain)")

    async def shutdown(self) -> None:
        self._running = False
        self.logger.info("LLMManager desligado")

    async def health_check(self) -> dict:
        return {
            "running": self._running,
            "model": self._model,
            "ollama_available": await self._check_ollama(),
            "history_size": 0,
        }

"""Extrator de conhecimento."""

from typing import Dict, Any, List, Optional
from datetime import datetime
import re


class KnowledgeExtractor:
    """
    Extrai conhecimento de conversas e interações.
    """
    
    def __init__(self, learning_manager):
        self.learning_manager = learning_manager
        self.logger = learning_manager.logger
        self.knowledge_manager = learning_manager.knowledge_manager
        
        self._extracted: List[Dict[str, Any]] = []
    
    async def extract_from_text(self, text: str, source: str = "unknown") -> List[Dict[str, Any]]:
        """
        Extrai conhecimento de um texto.
        """
        knowledge = []
        
        # 1. Extrai definições
        definitions = self._extract_definitions(text)
        for concept, definition in definitions:
            knowledge.append({
                "type": "definition",
                "concept": concept,
                "description": definition,
                "source": source
            })
        
        # 2. Extrai fatos
        facts = self._extract_facts(text)
        for fact in facts:
            knowledge.append({
                "type": "fact",
                "concept": fact["subject"],
                "description": fact["predicate"],
                "source": source
            })
        
        # 3. Extrai procedimentos
        procedures = self._extract_procedures(text)
        for procedure in procedures:
            knowledge.append({
                "type": "procedure",
                "name": procedure["name"],
                "steps": procedure["steps"],
                "source": source
            })
        
        # Armazena no KnowledgeManager
        for item in knowledge:
            if item["type"] == "definition":
                await self.knowledge_manager._on_add(
                    self._create_event({
                        "concept": item["concept"],
                        "description": item["description"],
                        "source": item["source"],
                        "confidence": 0.5
                    })
                )
        
        self._extracted.extend(knowledge)
        return knowledge
    
    async def extract_from_dict(self, data: Dict[str, Any], source: str = "unknown") -> List[Dict[str, Any]]:
        """
        Extrai conhecimento de um dicionário.
        """
        knowledge = []
        
        if "concept" in data and "description" in data:
            knowledge.append({
                "type": "definition",
                "concept": data["concept"],
                "description": data["description"],
                "source": source
            })
        
        if "facts" in data:
            for fact in data["facts"]:
                knowledge.append({
                    "type": "fact",
                    "concept": fact.get("subject", ""),
                    "description": fact.get("predicate", ""),
                    "source": source
                })
        
        return knowledge
    
    async def extract(self, limit: int = 10) -> List[Dict[str, Any]]:
        """
        Retorna conhecimento extraído recentemente.
        """
        return self._extracted[-limit:]
    
    def _extract_definitions(self, text: str) -> List[tuple]:
        """
        Extrai definições no formato "X é Y".
        """
        definitions = []
        
        # Padrões: "X é Y" ou "X significa Y" ou "X é quando Y"
        patterns = [
            r'([A-Za-zÀ-ÿ\s]{2,30})\s+[ée]\s+([A-Za-zÀ-ÿ\s,;]{5,100})',
            r'([A-Za-zÀ-ÿ\s]{2,30})\s+significa\s+([A-Za-zÀ-ÿ\s,;]{5,100})',
            r'([A-Za-zÀ-ÿ\s]{2,30})\s+[ée]\s+quando\s+([A-Za-zÀ-ÿ\s,;]{5,100})'
        ]
        
        for pattern in patterns:
            matches = re.findall(pattern, text)
            for match in matches:
                concept = match[0].strip()
                definition = match[1].strip()
                if len(concept) > 2 and len(definition) > 10:
                    definitions.append((concept, definition))
        
        return definitions[:3]
    
    def _extract_facts(self, text: str) -> List[Dict[str, str]]:
        """
        Extrai fatos no formato "X é Y" ou "X tem Y".
        """
        facts = []
        
        patterns = [
            r'([A-Za-zÀ-ÿ\s]{2,30})\s+[ée]\s+([A-Za-zÀ-ÿ\s,;]{3,50})',
            r'([A-Za-zÀ-ÿ\s]{2,30})\s+tem\s+([A-Za-zÀ-ÿ\s,;]{3,50})',
        ]
        
        for pattern in patterns:
            matches = re.findall(pattern, text)
            for match in matches:
                facts.append({
                    "subject": match[0].strip(),
                    "predicate": match[1].strip()
                })
        
        return facts[:5]
    
    def _extract_procedures(self, text: str) -> List[Dict[str, Any]]:
        """
        Extrai procedimentos (passos).
        """
        procedures = []
        
        # Detecta listas de passos
        if "primeiro" in text and "depois" in text:
            steps = re.findall(r'(?:primeiro|depois|em seguida|então|finalmente)\s+([A-Za-zÀ-ÿ\s,;]{5,50})', text)
            if steps:
                procedures.append({
                    "name": "Procedimento extraído",
                    "steps": steps
                })
        
        return procedures
    
    def _create_event(self, data: Dict[str, Any]) -> Any:
        """Cria um evento para o KnowledgeManager."""
        from src.kernel.events import Event
        return Event(
            type="knowledge.add",
            payload=data,
            source="knowledge_extractor"
        )
