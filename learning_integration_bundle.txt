#!/usr/bin/env python3
"""EVA OS - Ponto de entrada principal."""
import sys
import os
sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))
import asyncio
import signal
from pathlib import Path
import threading
import time
import socket

sys.path.insert(0, str(Path(__file__).parent))

from src.kernel import Boot


def check_ollama():
    try:
        sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
        sock.settimeout(1)
        result = sock.connect_ex(('127.0.0.1', 11434))
        sock.close()
        return result == 0
    except:
        return False


class EVAOS:
    def __init__(self):
        self.boot = Boot()
        self._shutdown_requested = False
        self._shutdown_lock = asyncio.Lock()
        self._shutdown_event = asyncio.Event()
        self._llm = None
        self._brain = None
        self._web_server = None
        self._console_thread = None
        self._loop = None
        self.logger = None
        
        # Referências para serviços (para shutdown)
        self._memory_manager = None
        self._long_term_memory = None
        self._lifecycle_manager = None   # ← NOVO
    
    def _run_console(self) -> None:
        """Inicia o console interativo (cmd) em uma thread separada.

        A thread só bloqueia em input(); publicação de eventos usa o loop principal.
        """
        try:
            from src.interfaces.cli.console import EVAConsole
            
            core = self.boot.core
            if not core:
                print("❌ Core não inicializado")
                return
            
            if not self._loop:
                print("❌ Event loop principal não disponível para o console")
                return
            
            console = EVAConsole(
                core.event_bus, 
                core.logger, 
                core.config,
                loop=self._loop,
                shutdown_event=self._shutdown_event,
                lifecycle_manager=self._lifecycle_manager,
                brain=self._brain,
            )
            
            try:
                console.cmdloop()
            except KeyboardInterrupt:
                pass
            
        except Exception as e:
            print(f"❌ Erro no console: {e}")
            import traceback
            traceback.print_exc()
            if self._loop and self._loop.is_running():
                self._loop.call_soon_threadsafe(self._shutdown_event.set)
            else:
                self._shutdown_event.set()
    
    async def run(self) -> None:
        signal.signal(signal.SIGINT, self._signal_handler)
        signal.signal(signal.SIGTERM, self._signal_handler)
        
        print("\n🔍 Verificando Ollama...")
        if check_ollama():
            print("✅ Ollama está rodando!")
        else:
            print("⚠️ Ollama não detectado.")
        
        success = await self.boot.boot()
        
        if not success:
            print("❌ Falha no boot. Encerrando.")
            sys.exit(1)
        
        core = self.boot.core
        self.logger = core.logger
        registry = core.service_registry
        event_bus = core.event_bus
        logger = core.logger
        config = core.config
        self._loop = asyncio.get_running_loop()

        # ============================================================
        # Obtém o LifecycleManager do core (já deve existir)
        # ============================================================
        self._lifecycle_manager = getattr(core, 'lifecycle_manager', None)
        if not self._lifecycle_manager:
            self.logger.warning("LifecycleManager não encontrado no core; shutdown será manual.")
        
        # ============================================================
        # 1. LLM
        # ============================================================
        print("🧠 Iniciando LLM...")
        try:
            from src.llm.llm_manager import LLMManager
            self._llm = LLMManager(event_bus, logger, config)
            await self._llm.initialize()
            registry.register("llm_manager", self._llm)
            print("✅ LLM iniciado")
        except Exception as e:
            self._llm = None
            print(f"❌ Erro ao iniciar LLM: {e}")
            import traceback
            traceback.print_exc()
        
        # ============================================================
        # 2. BRAIN
        # ============================================================
        print("🧠 Iniciando Brain...")
        try:
            from src.consciousness.brain import Brain
            self._brain = Brain(event_bus, logger, config, service_registry=registry)
            self._brain._llm_manager = self._llm
            await self._brain.initialize()
            registry.register("brain", self._brain)

            # EVA_ETAPA_33_3_VINCULACAO_POS_BRAIN
            # Vincula somente agentes ja iniciados pelo AgentManager.
            # Nao cria novas instancias nem reinicia os plugins existentes.
            plugin_manager = core.plugin_manager
            owner = registry.get("agent_manager")
            for plugin_name in (
                "knowledge_agent",
                "memory_agent",
                "planner_agent",
            ):
                plugin_id = plugin_manager._plugins_by_name.get(plugin_name)
                plugin = plugin_manager._plugins.get(plugin_id)

                if plugin is None or owner is None:
                    logger.warning(
                        "VINCULACAO_33_3_INDISPONIVEL: " + plugin_name
                    )
                    continue

                success = await plugin_manager._load_plugin(plugin)
                expected = owner.get_agent(
                    plugin_name.removesuffix("_agent")
                )
                same_instance = (
                    expected is not None
                    and registry.get(plugin_name) is expected
                    and plugin.instance is expected
                )

                if success and same_instance:
                    logger.info(
                        "VINCULACAO_33_3_OK: " + plugin_name
                    )
                else:
                    logger.error(
                        "VINCULACAO_33_3_FALHOU: "
                        + plugin_name
                        + " - "
                        + str(plugin.error)
                    )
            self._memory_manager = self._brain._modules.get("memory_manager")
            self._long_term_memory = self._brain._modules.get("long_term_memory")
            # EVA_ETAPA_34_3_CONVERSAS_COMPARTILHADAS
            # Reutiliza exclusivamente as instancias criadas pelo Brain.
            try:
                from pathlib import Path as _ConversationPath
                from datetime import datetime as _ConversationDateTime
                from src.core.plugin_manager import PluginStatus as _ConversationStatus

                _coordinator = getattr(
                    self._brain, "conversation_coordinator", None
                )
                if _coordinator is None:
                    raise RuntimeError("Coordenador do Brain indisponivel")

                _components = (
                    ("conversation_coordinator", _coordinator),
                    ("conversation_archive", _coordinator.archive),
                    ("turn_state_manager", _coordinator.states),
                )

                _expected_db = _ConversationPath(
                    _coordinator.db_path
                ).resolve()
                _validated = []

                # Valida todos antes de registrar qualquer componente.
                for _name, _instance in _components:
                    _plugin = core.plugin_manager.get_plugin(_name)
                    if _plugin is None and _instance is not None:
                        _instance_type = type(_instance)
                        _plugin = next(
                            (
                                _candidate
                                for _candidate in core.plugin_manager._plugins.values()
                                if _candidate.manifest.main_class == _instance_type.__name__
                                and _candidate.module is not None
                                and getattr(
                                    _candidate.module,
                                    _candidate.manifest.main_class,
                                    None,
                                ) is _instance_type
                            ),
                            None,
                        )
                    if _plugin is None or _instance is None:
                        raise RuntimeError("Plugin ou instancia ausente: " + _name)

                    _class = getattr(
                        _plugin.module,
                        _plugin.manifest.main_class,
                        None,
                    )
                    if not isinstance(_class, type):
                        raise RuntimeError("Classe nao encontrada: " + _name)

                    if not isinstance(_instance, _class):
                        raise RuntimeError("Classe divergente: " + _name)

                    if _ConversationPath(_instance.db_path).resolve() != _expected_db:
                        raise RuntimeError("Banco divergente: " + _name)

                    for _dependency in _plugin.manifest.requires:
                        if core.plugin_manager._resolve_service(_dependency) is None:
                            raise RuntimeError(
                                "Dependencia indisponivel: "
                                + _name + "/" + _dependency
                            )

                    if _plugin.instance is not None and _plugin.instance is not _instance:
                        raise RuntimeError("Outra instancia no plugin: " + _name)

                    _existing = registry.get(_name)
                    if _existing is not None and _existing is not _instance:
                        raise RuntimeError("Outra instancia no registro: " + _name)

                    _validated.append((_name, _instance, _plugin))

                for _name, _instance, _plugin in _validated:
                    if registry.get(_name) is None:
                        registry.register(
                            _name, _instance, _plugin.manifest.requires
                        )

                    _plugin.instance = _instance
                    _plugin.status = _ConversationStatus.STARTED
                    _plugin.error = None

                    if _plugin.loaded_at is None:
                        _plugin.loaded_at = _ConversationDateTime.now()
                    if _plugin.initialized_at is None:
                        _plugin.initialized_at = _ConversationDateTime.now()

                    _plugin.started_at = _ConversationDateTime.now()

                    # O Brain continua responsavel pelo ciclo de vida.
                    core.plugin_manager._delegated_plugins[
                        _plugin.manifest.id
                    ] = self._brain

                    logger.info("VINCULACAO_34_3_OK: " + _name)

            except Exception:
                logger.exception("VINCULACAO_34_3_FALHOU")
            print("✅ Brain iniciado")
        except Exception as e:
            print(f"❌ Erro ao iniciar brain: {e}")
            import traceback
            traceback.print_exc()
        
        # ============================================================
        # 3. WEB SERVER
        # ============================================================
        print("🌐 Iniciando Web Server...")
        try:
            from src.interfaces.web.web_server import WebServer
            self._web_server = WebServer(event_bus, logger, config)
            await self._web_server.initialize()
            await self._web_server.start()
            registry.register("web_server", self._web_server)
            print("✅ Web Server iniciado")
        except Exception as e:
            print(f"❌ Erro ao iniciar Web Server: {e}")
            import traceback
            traceback.print_exc()
        
        # ============================================================
        # 4. CONSOLE (cmd) - em thread separada
        # ============================================================
        print("💬 Iniciando Console...")
        try:
            self._console_thread = threading.Thread(target=self._run_console, daemon=False)
            self._console_thread.start()
            time.sleep(1)
            print("✅ Console iniciado")
        except Exception as e:
            print(f"❌ Erro ao iniciar Console: {e}")
            self._shutdown_event.set()
        
        print("\n" + "=" * 60)
        print("🚀 EVA OS PRONTA!")
        print("   💬 Console: digite qualquer coisa para conversar")
        print("   🌐 Web: http://localhost:8000")
        print("   🎤 Fale com a EVA usando o microfone")
        print("=" * 60 + "\n")
        # ========== ALTERADO: mensagem de encerramento ==========
        print("💡 Digite 'eva encerrar sistema' para encerrar\n")
        # =======================================================
        
        # ============================================================
        # 5. LOOP PRINCIPAL – AGUARDA EVENTO DE SHUTDOWN
        # ============================================================
        try:
            await self._shutdown_event.wait()
            if self.logger:
                self.logger.info("📴 Evento de shutdown detectado")
            else:
                print("📴 Evento de shutdown detectado")
        except asyncio.CancelledError:
            if self.logger:
                self.logger.info("📴 Loop principal cancelado")
            else:
                print("📴 Loop principal cancelado")
        finally:
            await self.shutdown()
    
    def _signal_handler(self, signum, frame) -> None:
        print(f"\n📴 Desligando (sinal)...")
        if self._loop and self._loop.is_running():
            self._loop.call_soon_threadsafe(self._shutdown_event.set)
        else:
            self._shutdown_event.set()
    
    async def shutdown(self) -> None:
        """Desliga a EVA de forma segura e idempotente."""
        async with self._shutdown_lock:
            if self._shutdown_requested:
                return
            self._shutdown_requested = True
            self._shutdown_event.set()
            
            if self.logger:
                self.logger.info("=" * 50)
                self.logger.info("EVA OS - Desligando...")
                self.logger.info("=" * 50)
            else:
                print("=" * 50)
                print("EVA OS - Desligando...")
                print("=" * 50)
            
            # ----------------------------------------------------------
            # 1. Brain primeiro (desliga módulos vivos, inclusive MemoryManager)
            # ----------------------------------------------------------
            if self._brain:
                try:
                    await self._brain.shutdown()
                    print("✅ Brain desligado")
                except Exception as e:
                    print(f"❌ Erro ao desligar Brain: {e}")
                    if self.logger:
                        self.logger.error(f"Erro ao desligar Brain: {e}")
            
            # ----------------------------------------------------------
            # 2. Persistência extra da mesma instância registrada no Core
            # ----------------------------------------------------------
            try:
                registry = self.boot.core.service_registry if self.boot.core else None
                memory_manager = None
                if registry:
                    memory_manager = registry.get("memory_manager")
                if memory_manager is None:
                    memory_manager = self._memory_manager
                if memory_manager and hasattr(memory_manager, "_save_memories"):
                    await memory_manager._save_memories()
                    print("✅ Memória salva (MemoryManager)")
                    if self.logger:
                        self.logger.info("Memória salva (MemoryManager)")
                # LongTermMemory
                long_term = registry.get("long_term_memory") if registry else self._long_term_memory
                if long_term and hasattr(long_term, "_save"):
                    await long_term._save()
                    print("✅ Memória de longo prazo salva")
                # KnowledgeManager
                knowledge = registry.get("knowledge_manager") if registry else None
                if knowledge and hasattr(knowledge, "_save_knowledge"):
                    await knowledge._save_knowledge()
                    print("✅ Conhecimento salvo")
            except Exception as e:
                if self.logger:
                    self.logger.error(f"Erro ao salvar memória: {e}")
                else:
                    print(f"Erro ao salvar memória: {e}")
            
            # ----------------------------------------------------------
            # 3. Demais componentes
            # ----------------------------------------------------------
            if self._web_server:
                try:
                    await self._web_server.shutdown()
                    print("✅ Web Server desligado")
                except Exception as e:
                    print(f"❌ Erro ao desligar Web Server: {e}")
            
            if self._lifecycle_manager:
                try:
                    await self._lifecycle_manager.stop_all()
                    print("✅ LifecycleManager.stop_all concluído")
                except Exception as e:
                    print(f"❌ Erro ao parar módulos via LifecycleManager: {e}")
            
            # ----------------------------------------------------------
            # Desliga o Boot (core, plugins, health, eventbus)
            # ----------------------------------------------------------
            try:
                await self.boot.shutdown()
                print("✅ Boot desligado")
            except Exception as e:
                print(f"❌ Erro ao desligar Boot: {e}")
            
            # ----------------------------------------------------------
            # Aguarda a thread do console terminar
            # ----------------------------------------------------------
            if self._console_thread and self._console_thread.is_alive():
                try:
                    self._console_thread.join(timeout=2)
                    if self._console_thread.is_alive():
                        if self.logger:
                            self.logger.warning("Thread do console não terminou a tempo")
                        else:
                            print("⚠️ Thread do console não terminou a tempo")
                except Exception as e:
                    if self.logger:
                        self.logger.error(f"Erro ao aguardar console: {e}")
            
            if self.logger:
                self.logger.info("✅ EVA OS desligado com sucesso")
            print("✅ EVA OS desligado com sucesso")
            print("=" * 50)


async def main():
    eva = EVAOS()
    await eva.run()

    def select_agent(self, intent: str, task: Dict[str, Any] = None) -> Optional[str]:
        """
        Seleciona o melhor agente para uma tarefa.
        
        Estratégia:
        1. Agentes especializados na intenção
        2. Agentes com capacidades correspondentes
        3. Agente geral (fallback)
        """
        # Mapeamento de intenção para agente
        intent_map = {
            "conversation": "general",
            "question": "knowledge",
            "command": "information",
            "planning": "planner",
            "learning": "learning",
            "emergency": "security",
            "memory": "memory",
            "home_automation": "home_automation",
            "information": "information"
        }
        
        preferred = intent_map.get(intent)
        
        if preferred and preferred in self._agents:
            agent = self._agents[preferred]
            if agent.can_handle(intent) and self._is_available(preferred):
                return preferred
        
        # Busca agente por capacidade
        for name, agent in self._agents.items():
            if agent.can_handle(intent) and self._is_available(name):
                return name
        
        # Fallback: general agent
        if "general" in self._agents and self._is_available("general"):
            return "general"
        
        # Qualquer agente disponível
        for name in self._agents.keys():
            if self._is_available(name):
                return name
        
        return None
    
    def _is_available(self, agent_name: str) -> bool:
        """Verifica se um agente está disponível."""
        status = self._agent_status.get(agent_name, {})
        return status.get("status") != "busy"

    async def _on_assign_task(self, event: Event) -> None:
        """Atribui uma tarefa a um agente."""
        payload = event.payload
        agent_name = payload.get("agent")
        task = payload.get("task")
        request_id = payload.get("request_id")
        
        if not agent_name or not task:
            await self._send_error("invalid_request", "Agente ou tarefa não fornecidos", request_id)
            return
        
        if agent_name not in self._task_queues:
            await self._send_error("agent_not_found", f"Agente {agent_name} não encontrado", request_id)
            return
        
        # Adiciona à fila
        await self._task_queues[agent_name].put({
            "task": task,
            "request_id": request_id,
            "timestamp": datetime.now().isoformat()
        })
        
        # Atualiza status
        self._agent_status[agent_name]["status"] = "queued"
        
        self.logger.debug(f"Tarefa enfileirada para {agent_name}")
    
    async def _process_queue(self, agent_name: str) -> None:
        """Processa a fila de um agente."""
        queue = self._task_queues.get(agent_name)
        agent = self._agents.get(agent_name)
        
        if not queue or not agent:
            return
        
        while self._running:
            try:
                # Pega da fila
                item = await queue.get()
                
                # Atualiza status
                self._agent_status[agent_name]["status"] = "busy"
                
                # Processa
                try:
                    await agent.handle_event(
                        Event(
                            type=f"agent.{agent_name}",
                            payload=item,
                            source="agent_manager"
                        )
                    )
                    self._agent_status[agent_name]["task_count"] = \
                        self._agent_status[agent_name].get("task_count", 0) + 1
                except Exception as e:
                    self.logger.error(f"Erro processando tarefa em {agent_name}: {e}")
                
                # Atualiza status
                self._agent_status[agent_name]["status"] = "idle"
                
                queue.task_done()
                
            except asyncio.CancelledError:
                break
            except Exception as e:
                self.logger.error(f"Erro no worker de {agent_name}: {e}")

    async def initialize(self) -> None:
        """Inicializa o AgentManager."""
        self.logger.info("AgentManager inicializando...")
        
        # Registra agentes base
        self._register_base_agents()
        
        self._running = True
        
        # Inicia monitoramento
        self._monitor_task = asyncio.create_task(self._monitor_agents())
        
        self.logger.info(f"AgentManager inicializado com {len(self._agents)} agentes")
    
    async def start(self) -> None:
        """Aguarda a inicializacao antes de iniciar agentes e workers."""
        async with self._start_lock:
            if self._shutting_down:
                raise RuntimeError("AgentManager em desligamento")
            for name, agent in list(self._agents.items()):
                if self._shutting_down:
                    return
                if name in self._started_agents:
                    continue
                initialization = self._initialization_tasks.get(name)
                if initialization is None:
                    self._agent_status[name]["status"] = "initialization_failed"
                    self.logger.error("Inicializacao nao rastreada: " + name)
                    continue
                try:
                    initialized = await asyncio.shield(initialization)
                except asyncio.CancelledError:
                    if self._shutting_down:
                        return
                    if initialization.cancelled():
                        continue
                    raise
                if self._shutting_down:
                    return
                if not initialized:
                    continue
                try:
                    await agent.start()
                except asyncio.CancelledError:
                    self._agent_status[name]["status"] = "start_cancelled"
                    raise
                except Exception as exc:
                    self._agent_status[name]["status"] = "start_failed"
                    self._agent_status[name]["error_type"] = type(exc).__name__
                    self.logger.error(
                        "Erro ao iniciar agente " + name + ": " + type(exc).__name__
                    )
                    continue
                if self._shutting_down:
                    return
                self._started_agents.add(name)
                self._agent_status[name]["status"] = "idle"
                worker = self._queue_workers.get(name)
                if worker is None or worker.done():
                    self._queue_workers[name] = asyncio.create_task(
                        self._process_queue(name)
                    )
            self.logger.info(
                "AgentManager: agentes iniciados=" + str(len(self._started_agents))
            )

    
    async def shutdown(self) -> None:
        """Cancela inicializacoes pendentes antes de desligar agentes."""
        self._shutting_down = True
        self._running = False
        initializations = list(self._initialization_tasks.values())
        for task in initializations:
            if not task.done():
                task.cancel()
        if initializations:
            await asyncio.gather(*initializations, return_exceptions=True)
        async with self._start_lock:
            if self._shutdown_complete:
                return
            workers = list(self._queue_workers.values())
            for task in workers:
                if not task.done():
                    task.cancel()
            if workers:
                await asyncio.gather(*workers, return_exceptions=True)
            if self._monitor_task is not None:
                self._monitor_task.cancel()
                await asyncio.gather(self._monitor_task, return_exceptions=True)
                self._monitor_task = None
            for name, agent in list(self._agents.items()):
                try:
                    await agent.shutdown()
                except Exception as exc:
                    self._agent_status[name]["status"] = "shutdown_failed"
                    self._agent_status[name]["error_type"] = type(exc).__name__
                    self.logger.error(
                        "Erro ao desligar agente " + name + ": " + type(exc).__name__
                    )
                else:
                    self._agent_status[name]["stopped"] = True
            self._started_agents.clear()
            self._shutdown_complete = True
            self.logger.info("AgentManager desligado")

    def register_agent(self, agent_class: Type[BaseAgent]) -> bool:
        """Rastreia a inicializacao; worker inicia somente apos start."""
        if self._shutting_down:
            return False
        if any(type(agent) is agent_class for agent in self._agents.values()):
            self.logger.warning("AgentManager: classe de agente ja registrada")
            return False
        try:
            asyncio.get_running_loop()
            agent = agent_class(
                event_bus=self.event_bus, logger=self.logger, config=self.config
            )
            agent.service_registry = self.service_registry
            if agent.name in self._agents:
                self.logger.warning("AgentManager: nome de agente ja registrado")
                return False
            self._agents[agent.name] = agent
            self._agent_status[agent.name] = {
                "status": "initializing",
                "capabilities": agent.capabilities,
                "task_count": 0,
            }
            self._task_queues[agent.name] = asyncio.Queue()
            self._initialization_tasks[agent.name] = asyncio.create_task(
                self._initialize_agent(agent.name, agent)
            )
            self.logger.info("Agente registrado: " + agent.name)
            return True
        except Exception as exc:
            self.logger.error("Erro ao registrar agente: " + type(exc).__name__)
            return False

    
    async def _on_register(self, event: Event) -> None:
        """Registra um agente via evento."""
        payload = event.payload
        agent_class_path = payload.get("agent_class")
        
        if not agent_class_path:
            self.logger.error("Nenhuma classe de agente fornecida")
            return
        
        try:
            # Importa dinamicamente
            module_path, class_name = agent_class_path.rsplit(".", 1)
            module = __import__(module_path, fromlist=[class_name])
            agent_class = getattr(module, class_name)
            
            success = self.register_agent(agent_class)
            
            await self.event_bus.publish_async(
                "agent.registered",
                {
                    "success": success,
                    "agent_class": agent_class_path,
                    "timestamp": datetime.now().isoformat()
                },
                source="agent_manager"
            )
            
        except Exception as e:
            self.logger.error(f"Erro ao registrar agente via evento: {e}")

    def select_agent(self, intent: str, task: Dict[str, Any] = None) -> Optional[str]:
        """
        Seleciona o melhor agente para uma tarefa.
        
        Estratégia:
        1. Agentes especializados na intenção
        2. Agentes com capacidades correspondentes
        3. Agente geral (fallback)
        """
        # Mapeamento de intenção para agente
        intent_map = {
            "conversation": "general",
            "question": "knowledge",
            "command": "information",
            "planning": "planner",
            "learning": "learning",
            "emergency": "security",
            "memory": "memory",
            "home_automation": "home_automation",
            "information": "information"
        }
        
        preferred = intent_map.get(intent)
        
        if preferred and preferred in self._agents:
            agent = self._agents[preferred]
            if agent.can_handle(intent) and self._is_available(preferred):
                return preferred
        
        # Busca agente por capacidade
        for name, agent in self._agents.items():
            if agent.can_handle(intent) and self._is_available(name):
                return name
        
        # Fallback: general agent
        if "general" in self._agents and self._is_available("general"):
            return "general"
        
        # Qualquer agente disponível
        for name in self._agents.keys():
            if self._is_available(name):
                return name
        
        return None
    
    def _is_available(self, agent_name: str) -> bool:
        """Somente agentes iniciados podem ser selecionados."""
        if self._shutting_down or agent_name not in self._started_agents:
            return False
        return self._agent_status.get(agent_name, {}).get("status") not in (
            "busy", "initialization_failed", "initialization_cancelled",
            "start_failed", "start_cancelled", "shutdown_failed",
        )

    
    # ===== ATRIBUIÇÃO =====
    
    async def _on_assign_task(self, event: Event) -> None:
        """Atribui uma tarefa a um agente."""
        payload = event.payload
        agent_name = payload.get("agent")
        task = payload.get("task")
        request_id = payload.get("request_id")
        
        if not agent_name or not task:
            await self._send_error("invalid_request", "Agente ou tarefa não fornecidos", request_id)
            return
        
        if agent_name not in self._task_queues:
            await self._send_error("agent_not_found", f"Agente {agent_name} não encontrado", request_id)
            return
        
        # Adiciona à fila
        await self._task_queues[agent_name].put({
            "task": task,
            "request_id": request_id,
            "timestamp": datetime.now().isoformat()
        })
        
        # Atualiza status
        self._agent_status[agent_name]["status"] = "queued"
        
        self.logger.debug(f"Tarefa enfileirada para {agent_name}")
    
    async def _process_queue(self, agent_name: str) -> None:
        """Processa a fila de um agente."""
        queue = self._task_queues.get(agent_name)
        agent = self._agents.get(agent_name)
        
        if not queue or not agent:
            return
        
        while self._running:
            try:
                # Pega da fila
                item = await queue.get()
                
                # Atualiza status
                self._agent_status[agent_name]["status"] = "busy"
                
                # Processa
                try:
                    await agent.handle_event(
                        Event(
                            type=f"agent.{agent_name}",
                            payload=item,
                            source="agent_manager"
                        )
                    )
                    self._agent_status[agent_name]["task_count"] = \
                        self._agent_status[agent_name].get("task_count", 0) + 1
                except Exception as e:
                    self.logger.error(f"Erro processando tarefa em {agent_name}: {e}")
                
                # Atualiza status
                self._agent_status[agent_name]["status"] = "idle"
                
                queue.task_done()
                
            except asyncio.CancelledError:
                break
            except Exception as e:
                self.logger.error(f"Erro no worker de {agent_name}: {e}")

            except asyncio.CancelledError:

            except Exception as e:
                self.logger.error(f"Erro no monitoramento: {e}")

"""Registro de serviços do Core."""

from enum import Enum, auto
from typing import Any, Dict, List, Optional, Type


class ServiceStatus(Enum):
    """Status de um serviço."""
    REGISTERED = auto()
    INITIALIZED = auto()
    STARTING = auto()
    RUNNING = auto()
    PAUSED = auto()
    STOPPING = auto()
    STOPPED = auto()
    FAILED = auto()


class Service:
    """Definição de um serviço registrado."""
    
    def __init__(self, name: str, instance: Any, dependencies: List[str] = None):
        self.name = name
        self.instance = instance
        self.dependencies = dependencies or []
        self.status = ServiceStatus.REGISTERED
        self.service_type = type(instance).__name__
        self.version = getattr(instance, "version", "0.0.1")
    
    def __repr__(self):
        return f"Service({self.name}, {self.service_type}, {self.status.name})"


class ServiceRegistry:
    """
    Catálogo central da EVA.
    Responde: Onde está o MemoryManager? Onde está o Planner?
    """
    
    def __init__(self, logger):
        self.logger = logger
        self._services: Dict[str, Service] = {}
        self._aliases: Dict[str, str] = {}
    
    def register(self, name: str, instance: Any, dependencies: List[str] = None) -> None:
        """
        Registra um serviço.
        
        Args:
            name: Nome único do serviço
            instance: Instância do serviço
            dependencies: Lista de nomes de serviços dos quais depende
        """
        if name in self._services:
            self.logger.warning(f"Serviço {name} já registrado. Substituindo...")
        
        service = Service(name, instance, dependencies)
        self._services[name] = service
        self.logger.info(f"Serviço registrado: {name} ({service.service_type})")
    
    def register_alias(self, alias: str, target: str) -> None:
        """Registra um alias para um serviço."""
        if target not in self._services:
            raise ValueError(f"Serviço alvo {target} não encontrado")
        self._aliases[alias] = target
        self.logger.debug(f"Alias registrado: {alias} -> {target}")
    
    def unregister(self, name: str) -> bool:
        """Remove um serviço do registro."""
        if name in self._services:
            del self._services[name]
            # Remove aliases
            self._aliases = {k: v for k, v in self._aliases.items() if v != name}
            self.logger.info(f"Serviço removido: {name}")
            return True
        return False
    
    def get(self, name: str) -> Optional[Any]:
        """
        Obtém um serviço pelo nome.
        Suporta aliases.
        """
        # Resolve alias
        actual_name = self._aliases.get(name, name)
        
        service = self._services.get(actual_name)
        if service:
            return service.instance
        return None
    
    def get_service(self, name: str) -> Optional[Service]:
        """Obtém o objeto Service completo."""
        actual_name = self._aliases.get(name, name)
        return self._services.get(actual_name)
    
    def exists(self, name: str) -> bool:
        """Verifica se um serviço existe."""
        actual_name = self._aliases.get(name, name)
        return actual_name in self._services
    
    def list(self) -> List[str]:
        """Lista todos os nomes de serviços registrados."""
        return list(self._services.keys())

    @staticmethod
    def _normalize(value: Any) -> str:
        text = unicodedata.normalize("NFKD", str(value).casefold())
        return "".join(c for c in text if not unicodedata.combining(c))

    def _extract_keywords(self, message: str) -> List[str]:
        clean = re.sub(r"[^\w\s]", " ", self._normalize(message))
        stopwords = {"a", "o", "e", "de", "da", "do", "que", "com", "para", "por",
                     "um", "uma", "em", "no", "na", "os", "as", "se", "mas", "como",
                     "quando", "onde", "porque", "entao", "isso", "aquilo", "este",
                     "esta", "esses", "essas", "qual", "sua", "seu", "meu", "minha",
                     "nosso", "nossa"}
        result = []
        for word in clean.split():
            if len(word) > 3 and word not in stopwords:
                for term in (word, word[:-1] if word.endswith("s") and len(word) > 4 else word):
                    if term not in result:
                        result.append(term)
        return result

    @staticmethod
    def _entity_names(entities: Any) -> List[str]:
        """Aceita lista do ContextEngine e dicionários legados."""
        names = []
        if isinstance(entities, list):
            for entity in entities:
                if isinstance(entity, dict) and isinstance(entity.get("name"), str):
                    names.append(entity["name"])
                elif isinstance(entity, str):
                    names.append(entity)
        elif isinstance(entities, dict):
            if isinstance(entities.get("name"), str):
                names.append(entities["name"])
            else:
                for value in entities.values():
                    if isinstance(value, str):
                        names.append(value)
                    elif isinstance(value, dict) and isinstance(value.get("name"), str):
                        names.append(value["name"])
        return [name.strip() for name in names if name.strip()]

    def _search_facts(self, keywords: List[str]) -> List[Dict[str, Any]]:
        facts = self._knowledge_base.get("facts", {})
        if not isinstance(facts, dict) or not keywords:
            return []
        results = []
        for key, fact in facts.items():
            normalized_key = self._normalize(key)
            score = 1.0 if any(term in normalized_key for term in keywords) else 0.0
            if not score:
                content = json.dumps(fact, ensure_ascii=False, default=str)
                if any(term in self._normalize(content) for term in keywords):
                    score = 0.8
            if score:
                results.append({"type": "fact", "key": key, "data": deepcopy(fact), "relevance": score})
        results.sort(key=lambda item: item["relevance"], reverse=True)
        return results[:5]

    def _search_concepts(self, keywords: List[str]) -> List[Dict[str, Any]]:
        concepts = self._knowledge_base.get("concepts", {})
        if not isinstance(concepts, dict) or not keywords:
            return []
        results = []
        for key, definition in concepts.items():
            key_match = any(term in self._normalize(key) for term in keywords)
            definition_match = any(term in self._normalize(definition) for term in keywords)
            if key_match or definition_match:
                results.append({"type": "concept", "key": key,
                                "definition": deepcopy(definition),
                                "relevance": 1.0 if key_match else 0.7})
        results.sort(key=lambda item: item["relevance"], reverse=True)
        return results[:3]

    def _search_entities(self, entities: Any) -> List[Dict[str, Any]]:
        facts = self._knowledge_base.get("facts", {})
        if not isinstance(facts, dict):
            return []
        results = []
        seen = set()
        for name in self._entity_names(entities):
            needle = self._normalize(name)
            for key, fact in facts.items():
                if key not in seen and needle in self._normalize(key):
                    results.append({"type": "entity", "entity_type": "name",
                                    "entity_value": name, "data": deepcopy(fact), "relevance": 1.0})
                    seen.add(key)
        return results[:3]

    def query(self, text: str) -> Dict[str, Any] | None:
        """Responde apenas perguntas factuais explícitas, com entidade e campo exatos.

        Uma correspondência de palavras-chave não constitui resposta confiável.
        Retorna None em casos ambíguos, compostos ou sem dados.
        """
        if not isinstance(text, str) or not text.strip():
            return None
        question = self._normalize(text)
        question = re.sub(r"[^\w\s]", " ", question)
        question = " ".join(question.split())
        # 'EVA' como vocativo não é a entidade consultada.
        question = re.sub(r"^eva\s+", "", question)

        patterns = (
            (r"(?:quem criou|quem e o criador (?:de|do|da)) (.+)", "criador", "{entity} foi criado por {value}."),
            (r"qual (?:e )?o criador (?:de|do|da) (.+)", "criador", "{entity} foi criado por {value}."),
            (r"qual (?:e )?a capital (?:de|do|da) (.+)", "capital", "A capital de {entity} é {value}."),
            (r"qual (?:e )?o significado (?:de|do|da) (.+)", "significado", "O significado de {entity} é {value}."),
            (r"(?:em que ano|qual ano) (.+) foi (?:criado|criada)", "ano", "{entity} foi criado em {value}."),
            (r"qual (?:e )?a moeda (?:de|do|da) (.+)", "moeda", "A moeda de {entity} é {value}."),
            (r"qual (?:e )?o idioma (?:de|do|da) (.+)", "idioma", "O idioma de {entity} é {value}."),
        )
        facts = self._knowledge_base.get("facts", {})
        if not isinstance(facts, dict):
            return None
        for pattern, field, template in patterns:
            match = re.fullmatch(pattern, question)
            if not match:
                continue
            requested = match.group(1).strip()
            if field == "criador":
                requested = re.sub(r"^(?:o|a)\s+", "", requested)
            candidates = [(key, record) for key, record in facts.items()
                          if self._normalize(key) == requested and isinstance(record, dict)]
            if len(candidates) != 1:
                return None
            key, record = candidates[0]
            value = record.get(field)
            if not isinstance(value, (str, int, float)) or isinstance(value, bool) or not str(value).strip():
                return None
            display = "EVA" if key == "eva" else key.capitalize()
            if key == "brasil" and field in {"capital", "moeda", "idioma"}:
                display = "do Brasil"
                template = template.replace("de {entity}", "{entity}")
            return {"answer": template.format(entity=display, value=value),
                    "confidence": 1.0, "resolved": True,
                    "source": "knowledge_base", "fact_key": key, "field": field}
        return None

    async def process(self, context: Dict[str, Any], memory: Dict[str, Any] = None) -> Dict[str, Any]:
        if not isinstance(context, dict):
            self.logger.warning("[KnowledgeManager] Contexto inválido; usando contexto vazio")
            context = {}
        message = context.get("message", "")
        if not isinstance(message, str):
            message = str(message) if message is not None else ""
        keywords = self._extract_keywords(message)
        entities = context.get("entities")
        if not entities and isinstance(context.get("state"), dict):
            entities = context["state"].get("active_entities", [])
        facts = self._search_facts(keywords)
        concepts = self._search_concepts(keywords)
        entity_knowledge = self._search_entities(entities)
        result = {"keywords": keywords, "facts": facts, "concepts": concepts,
                  "entity_knowledge": entity_knowledge,
                  "total_results": len(facts) + len(concepts) + len(entity_knowledge),
                  "timestamp": datetime.now().isoformat()}
        if self.event_bus is not None:
            await self.event_bus.publish_async(
                EventType.KNOWLEDGE_QUERIED,
                {"context": context, "memory": memory or {}, "knowledge": result,
                 "timestamp": datetime.now().isoformat()},
                source="knowledge_manager",
            )
        return result

    async def add_knowledge(self, key: str, data: Dict[str, Any], category: str = "facts") -> bool:
        if not isinstance(key, str) or not key.strip() or category not in ("facts", "concepts", "categories"):
            self.logger.warning("[KnowledgeManager] Chave ou categoria inválida")
            return False
        if not isinstance(data, (dict, str, list, int, float, bool)) and data is not None:
            self.logger.warning("[KnowledgeManager] Tipo de conhecimento inválido")
            return False
        section = self._knowledge_base.get(category)
        if not isinstance(section, dict):
            self.logger.error("[KnowledgeManager] Categoria corrompida: %s", category)
            return False
        key = key.strip()
        existed = key in section
        previous = deepcopy(section.get(key))
        section[key] = deepcopy(data)
        if not await self._save_knowledge():
            if existed:
                section[key] = previous
            else:
                section.pop(key, None)
            return False
        self.logger.info("[KnowledgeManager] Conhecimento adicionado: %s", key)
        return True

    async def _load_knowledge(self) -> None:
        if not self._knowledge_path.exists():
            return
        try:
            with self._knowledge_path.open("r", encoding="utf-8") as handle:
                loaded = json.load(handle)
            if not isinstance(loaded, dict):
                raise ValueError("O arquivo de conhecimento não contém um objeto JSON")
            for section in ("facts", "concepts", "categories"):
                value = loaded.get(section)
                if value is None:
                    continue
                if isinstance(value, dict):
                    self._knowledge_base[section].update(value)
                else:
                    self.logger.warning("[KnowledgeManager] Ignorando seção inválida: %s", section)
            self.logger.info("[KnowledgeManager] Conhecimento persistido carregado")
        except (OSError, json.JSONDecodeError, ValueError) as exc:
            self.logger.error("[KnowledgeManager] Falha ao carregar %s: %s", self._knowledge_path, exc)

    async def _save_knowledge(self) -> bool:
        temp_path = None
        try:
            self._knowledge_path.parent.mkdir(parents=True, exist_ok=True)
            with tempfile.NamedTemporaryFile(mode="w", encoding="utf-8", dir=self._knowledge_path.parent,
                                             prefix=".knowledge_", suffix=".tmp", delete=False) as handle:
                temp_path = Path(handle.name)
                json.dump(self._knowledge_base, handle, ensure_ascii=False, indent=2)
                handle.flush()
                os.fsync(handle.fileno())
            os.replace(temp_path, self._knowledge_path)
            return True
        except (OSError, TypeError, ValueError) as exc:
            self.logger.error("[KnowledgeManager] Falha ao salvar conhecimento: %s", exc)
            return False
        finally:
            if temp_path is not None:
                try:
                    temp_path.unlink(missing_ok=True)
                except OSError:
                    pass

    async def shutdown(self) -> None:
        self.logger.info("[KnowledgeManager] Desligando...")
        if not await self._save_knowledge():
            self.logger.error("[KnowledgeManager] Não foi possível persistir a base no desligamento")
        self._running = False

    async def health_check(self) -> dict:
        return {"status": "running" if self._running else "stopped",
                "facts": len(self._knowledge_base.get("facts", {})),
                "concepts": len(self._knowledge_base.get("concepts", {}))}
